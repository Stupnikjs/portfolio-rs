//! Outils de développement (hors métier) : concatène les sources du projet
//! dans un .txt à donner à un LLM.
//!
//! Sorti de `main.rs` : ce n'est pas du pipeline portefeuille.

use std::fs;
use std::path::{Path, PathBuf};

use anyhow::Result;

const SEPARATOR: &str = "################################################################################";

/// Fichiers d'extension `extension` sous `dir`, récursivement.
fn collect_files(dir: &Path, extension: &str, out: &mut Vec<PathBuf>) -> Result<()> {
    if !dir.exists() {
        return Ok(());
    }
    for entry in fs::read_dir(dir)? {
        let path = entry?.path();
        if path.is_dir() {
            collect_files(&path, extension, out)?;
        } else if path.extension().map_or(false, |ext| ext == extension) {
            out.push(path);
        }
    }
    Ok(())
}

/// (dossier, chemin relatif) affichés dans l'en-tête d'un fichier.
/// Un fichier hors de `root_dir` (Cargo.toml...) est rattaché à ".".
fn header_parts(path: &str, root_dir: &str) -> (String, String) {
    match path.strip_prefix(&format!("{root_dir}/")) {
        Some(rel) => {
            let dir = rel.rsplit_once('/').map_or_else(|| root_dir.to_string(), |(d, _)| format!("{root_dir}/{d}"));
            (dir, rel.to_string())
        }
        None => (".".to_string(), path.to_string()),
    }
}

fn render_file(path: &Path, root_dir: &str) -> Result<String> {
    let normalized = path.to_string_lossy().replace('\\', "/");
    let (dossier, relatif) = header_parts(&normalized, root_dir);
    let fichier = path.file_name().map(|f| f.to_string_lossy().into_owned()).unwrap_or_default();
    let content = fs::read_to_string(path)?;

    Ok(format!(
        "{SEPARATOR}\n# Dossier : {dossier}\n# Fichier : {fichier}\n# Chemin relatif : {relatif}\n{SEPARATOR}\n\n{content}\n\n"
    ))
}

/// Concatène `prepend_files` (dans l'ordre donné), puis les fichiers
/// `extension` de `root_dir` triés par chemin.
pub fn render_bundle(root_dir: &str, extension: &str, prepend_files: &[&str]) -> Result<String> {
    let mut files: Vec<PathBuf> = Vec::new();

    for prepend in prepend_files {
        let p = Path::new(prepend);
        if p.exists() {
            files.push(p.to_path_buf());
        } else {
            println!("[Omis] Fichier à inclure introuvable : {prepend}");
        }
    }

    let mut sources = Vec::new();
    if Path::new(root_dir).exists() {
        collect_files(Path::new(root_dir), extension, &mut sources)?;
    } else {
        println!("[Omis] Dossier introuvable pour la génération : {root_dir}");
    }
    sources.sort();
    files.extend(sources);

    let mut output = String::new();
    for file in &files {
        output.push_str(&render_file(file, root_dir)?);
    }
    Ok(output)
}

pub fn write_bundle(root_dir: &str, extension: &str, output_file: &str, prepend_files: &[&str]) -> Result<()> {
    fs::write(output_file, render_bundle(root_dir, extension, prepend_files)?)?;
    println!("{output_file} généré avec succès.");
    Ok(())
}

pub fn dump_for_llm() -> Result<()> {
    let mut rust_bundle = String::new();

    // 1. Tous les Cargo.toml du workspace (racine + crates)
    let cargo_files = [
        "Cargo.toml",
        "portfolio-rs/Cargo.toml", // Au cas où tu en as laissé un dans le sous-dossier
        "portfolio-rs/core/Cargo.toml",
        "portfolio-rs/pricer/Cargo.toml",
        "portfolio-rs/model/Cargo.toml",
        "portfolio-rs/parse/Cargo.toml",
    ];
    for file in &cargo_files {
        let path = Path::new(file);
        if path.exists() {
            rust_bundle.push_str(&render_file(path, ".")?);
        }
    }

    // 2. Tout le code source (.rs) de chaque crate
    for crate_name in &["core", "pricer", "model", "parse"] {
        let src_dir = format!("portfolio-rs/{crate_name}/src");
        if Path::new(&src_dir).exists() {
            rust_bundle.push_str(&render_bundle(&src_dir, "rs", &[])?);
        }
    }

    std::fs::write("forllm.txt", rust_bundle)?;
    println!("forllm.txt généré avec succès (Workspace complet).");

    // 3. Dashboard Python (si le dossier dashboard est aussi dans le dossier parent)
    let py_bundle = render_bundle("dashboard", "py", &[])?;
    std::fs::write("forllmdashboard.txt", py_bundle)?;
    println!("forllmdashboard.txt généré avec succès.");

    Ok(())
}

//! Portage de src/ledger/cost_basis.py -- calcul du cost basis (prix de
//! revient) par méthode FIFO, et du P&L réalisé à la vente.
//!
//! ⚠ FIFO = convention d'ANALYSE de ce projet (P&L latent / réalisé). Ce
//! n'est pas la règle fiscale des titres (prix moyen pondéré, PMP) : la
//! déclaration des plus-values du CTO doit passer par un calcul dédié.
//!
//! Le coût/produit de chaque transaction est dérivé de `value_eur` (champ
//! immutable qui fait autorité) plutôt que de `quantity * price` : `price`
//! est optionnel et parfois exprimé dans la devise locale de l'actif
//! plutôt qu'en EUR (cas XTB).
//!
//! Structure du fichier :
//! - types de résultat (`Lot`, `RealizedGain`, `CostBasisResult`) ;
//! - `consume_lots` : brique pure (une file de lots, une quantité) ;
//! - `compute_fifo` : rejoue les transactions et orchestre les briques.

use std::collections::{HashMap, VecDeque};

use anyhow::{anyhow, Result};
use chrono::{DateTime, Utc};

use crate::schema::{Transaction, TransactionKind};
use crate::store::serialize::TxStore;

const EPSILON: f64 = 1e-12;

// ---------------------------------------------------------------------
// Types de résultat
// ---------------------------------------------------------------------

/// Un lot FIFO : une tranche de quantité acquise à un prix donné.
#[derive(Debug, Clone)]
pub struct Lot {
    pub quantity: f64,
    pub unit_cost_eur: f64,
    pub acquired_at: DateTime<Utc>,
    pub external_id: Option<String>,
}

/// Résultat d'une vente rapprochée avec un ou plusieurs lots FIFO.
#[derive(Debug, Clone)]
pub struct RealizedGain {
    pub symbol: String,
    pub sell_time: DateTime<Utc>,
    pub quantity: f64,
    pub proceeds_eur: f64,
    pub cost_eur: f64,
    pub pnl_eur: f64,
    pub external_id: Option<String>,
    pub lots_consumed: Vec<Lot>,
    /// True si la vente dépasse les lots connus (transfert entrant non
    /// tracé, données historiques incomplètes...).
    pub incomplete: bool,
}

/// Sortie complète du calcul FIFO pour un wallet.
#[derive(Debug, Default)]
pub struct CostBasisResult {
    pub open_lots: HashMap<String, Vec<Lot>>,
    pub realized_gains: Vec<RealizedGain>,
    pub fees_eur_by_symbol: HashMap<String, f64>,
}

impl CostBasisResult {
    pub fn open_quantity(&self, symbol: &str) -> f64 {
        self.open_lots.get(symbol).map(|lots| lots.iter().map(|l| l.quantity).sum()).unwrap_or(0.0)
    }

    /// Coût de revient total des lots encore ouverts pour `symbol`.
    pub fn open_cost_basis(&self, symbol: &str) -> f64 {
        self.open_lots
            .get(symbol)
            .map(|lots| lots.iter().map(|l| l.quantity * l.unit_cost_eur).sum())
            .unwrap_or(0.0)
    }

    /// Prix de revient moyen par unité sur les lots ouverts (None si rien détenu).
    pub fn average_cost(&self, symbol: &str) -> Option<f64> {
        let qty = self.open_quantity(symbol);
        if qty <= 0.0 {
            return None;
        }
        Some(self.open_cost_basis(symbol) / qty)
    }

    pub fn total_realized_pnl(&self, symbol: Option<&str>) -> f64 {
        self.realized_gains
            .iter()
            .filter(|g| symbol.map_or(true, |s| g.symbol == s))
            .map(|g| g.pnl_eur)
            .sum()
    }

    pub fn total_fees(&self, symbol: Option<&str>) -> f64 {
        match symbol {
            Some(s) => *self.fees_eur_by_symbol.get(s).unwrap_or(&0.0),
            None => self.fees_eur_by_symbol.values().sum(),
        }
    }

    /// Cessions de TITRES uniquement (ni crypto, ni cash). Point d'entrée
    /// unique : remplace les filtres `stock_symbols` dupliqués dans
    /// main.rs, history.rs et realized.rs.
    ///
    /// Pourquoi exclure la crypto : les WITHDRAW (transferts entre
    /// wallets) sont vus comme des ventes FIFO et fausseraient le résultat.
    pub fn stock_realized_gains<'a>(&'a self, tx_store: &'a TxStore) -> impl Iterator<Item = &'a RealizedGain> + 'a {
        self.realized_gains.iter().filter(move |g| tx_store.is_stock(&g.symbol))
    }

    /// P&L réalisé total sur les titres.
    pub fn total_realized_pnl_stocks(&self, tx_store: &TxStore) -> f64 {
        self.stock_realized_gains(tx_store).map(|g| g.pnl_eur).sum()
    }
}

// ---------------------------------------------------------------------
// Briques pures
// ---------------------------------------------------------------------

fn unit_value_eur(quantity: f64, value_eur: f64) -> Result<f64> {
    if quantity <= 0.0 {
        return Err(anyhow!("quantity doit être positive, reçu {quantity}"));
    }
    Ok(value_eur / quantity)
}

/// Lot ouvert par un BUY / DEPOSIT.
fn open_lot(tx: &Transaction) -> Result<Lot> {
    Ok(Lot {
        quantity: tx.quantity,
        unit_cost_eur: unit_value_eur(tx.quantity, tx.value_eur)?,
        acquired_at: tx.time,
        external_id: tx.external_id.clone(),
    })
}

/// Ce qu'a coûté la consommation de `quantity` dans une file de lots.
#[derive(Debug)]
struct Consumption {
    /// Tranches réellement prélevées (une par lot touché).
    consumed: Vec<Lot>,
    cost_eur: f64,
    /// Quantité que les lots connus n'ont pas pu couvrir.
    uncovered: f64,
}

/// Prélève `quantity` dans `queue` en commençant par le lot le plus
/// ancien (FIFO). Pure : ne connaît ni les transactions ni les symboles.
fn consume_lots(queue: &mut VecDeque<Lot>, quantity: f64) -> Consumption {
    let mut remaining = quantity;
    let mut consumed = Vec::new();
    let mut cost_eur = 0.0;

    while remaining > EPSILON {
        let Some(lot) = queue.front_mut() else { break };
        let take = lot.quantity.min(remaining);

        cost_eur += take * lot.unit_cost_eur;
        consumed.push(Lot { quantity: take, ..lot.clone() });

        lot.quantity -= take;
        remaining -= take;
        if lot.quantity <= EPSILON {
            queue.pop_front();
        }
    }

    Consumption { consumed, cost_eur, uncovered: remaining.max(0.0) }
}

/// Transforme un SELL / WITHDRAW en plus-value réalisée.
///
/// Si la vente dépasse les lots connus, le reliquat est valorisé à un
/// coût de 0 et le résultat est marqué `incomplete` -- son `pnl_eur` est
/// donc surestimé d'autant (signalé plutôt qu'absorbé silencieusement).
fn realize(tx: &Transaction, queue: &mut VecDeque<Lot>) -> RealizedGain {
    let Consumption { mut consumed, cost_eur, uncovered } = consume_lots(queue, tx.quantity);

    let incomplete = uncovered > EPSILON;
    if incomplete {
        consumed.push(Lot { quantity: uncovered, unit_cost_eur: 0.0, acquired_at: tx.time, external_id: None });
    }

    RealizedGain {
        symbol: tx.asset.symbol.clone(),
        sell_time: tx.time,
        quantity: tx.quantity,
        proceeds_eur: tx.value_eur,
        cost_eur,
        pnl_eur: tx.value_eur - cost_eur,
        external_id: tx.external_id.clone(),
        lots_consumed: consumed,
        incomplete,
    }
}

// ---------------------------------------------------------------------
// Orchestration
// ---------------------------------------------------------------------

/// Rejoue les transactions du wallet dans l'ordre chronologique et
/// applique la méthode FIFO : chaque SELL/WITHDRAW consomme les lots
/// BUY/DEPOSIT les plus anciens en premier.
///
/// `at = None` -- comme pour `holdings_at` -- rejoue toutes les
/// transactions connues.
///
/// Les FEE consomment de la quantité (frais payés dans l'actif) mais ne
/// sont pas rattachés à un lot précis : accumulés séparément par actif
/// (`fees_eur_by_symbol`) -- à soustraire soi-même du P&L pour un
/// résultat net de frais. Un FEE qui dépasse les lots connus est ignoré
/// (edge case qui ne devrait pas arriver en pratique).
pub fn compute_fifo(tx_store: &TxStore, at: Option<DateTime<Utc>>) -> Result<CostBasisResult> {
    let mut lots: HashMap<String, VecDeque<Lot>> = HashMap::new();
    let mut realized: Vec<RealizedGain> = Vec::new();
    let mut fees: HashMap<String, f64> = HashMap::new();

    let mut transactions: Vec<&Transaction> = tx_store.transactions.iter().collect();
    transactions.sort_by_key(|tx| tx.time);

    let in_scope = |tx: &&Transaction| at.map_or(true, |at| tx.time <= at);

    for tx in transactions.into_iter().take_while(in_scope) {
        let queue = lots.entry(tx.asset.symbol.clone()).or_default();

        match tx.kind {
            TransactionKind::Buy | TransactionKind::Deposit => queue.push_back(open_lot(tx)?),
            TransactionKind::Sell | TransactionKind::Withdraw => realized.push(realize(tx, queue)),
            TransactionKind::Fee => {
                consume_lots(queue, tx.quantity);
                *fees.entry(tx.asset.symbol.clone()).or_insert(0.0) += tx.value_eur;
            }
        }
    }

    let open_lots: HashMap<String, Vec<Lot>> = lots
        .into_iter()
        .filter(|(_, queue)| !queue.is_empty())
        .map(|(symbol, queue)| (symbol, queue.into_iter().collect()))
        .collect();

    Ok(CostBasisResult { open_lots, realized_gains: realized, fees_eur_by_symbol: fees })
}

#[cfg(test)]

    fn approx(a: f64, b: f64) -> bool {
        (a - b).abs() < 1e-9
    }

    fn lot(quantity: f64, unit_cost_eur: f64) -> Lot {
        Lot { quantity, unit_cost_eur, acquired_at: utc("2025-01-01T00:00:00Z"), external_id: None }

    // --- consume_lots (pure, sans TxStore) ---

    #[test]
    fn consume_lots_takes_the_oldest_lot_first_and_splits_across_lots() {
        let mut queue: VecDeque<Lot> = vec![lot(10.0, 100.0), lot(10.0, 200.0)].into();
        let c = consume_lots(&mut queue, 15.0);

        assert!(approx(c.cost_eur, 10.0 * 100.0 + 5.0 * 200.0));
        assert_eq!(c.consumed.len(), 2);
        assert!(approx(c.uncovered, 0.0));
        assert_eq!(queue.len(), 1);
        assert!(approx(queue[0].quantity, 5.0)); // reste du 2e lot
    }

        assert!(approx(c.cost_eur, 500.0));
        assert!(approx(c.uncovered, 3.0));
        assert!(queue.is_empty());

        assert!(c.consumed.is_empty());
        assert!(approx(c.cost_eur, 0.0));

//! Historique de valorisation du portefeuille, hebdomadaire ou mensuel.
//!
//! Principe : un snapshot par période, pris en fin de journée (23:59:59 UTC) :
//! - `Frequency::Weekly`  : le DIMANCHE de chaque semaine ISO ;
//! - `Frequency::Monthly` : le PREMIER LUNDI de chaque mois.
//!
//! Les périodes passées sont immuables : on ne
//! (re)calcule jamais une date déjà présente dans history.json, SAUF si
//! l'entrée est périmée (ancien format sans détail par actif) ou
//! incomplète (au moins un prix manquant/reporté) -- elle est alors
//! recalculée au prochain run, ce qui permet de se rattraper après une
//! panne Yahoo/Binance. Chaque `cargo run` ne backfille que les dates
//! manquantes, et seulement celles dont la journée est terminée
//! (date < aujourd'hui) : la semaine / le mois en cours n'est pas écrit.
//!
//! Le P&L total d'une entrée = latent (valeur - cost basis des lots ouverts)
//! + plus-values RÉALISÉES CUMULÉES jusqu'à la date du snapshot. Sans le
//! réalisé, une vente fait disparaître son gain de toutes les courbes (le
//! produit part en cash, exclu de l'historique). Seules les actions
//! (`AssetKind::Stock`) comptent, comme dans `main.rs` : les WITHDRAW crypto
//! sont vus comme des ventes FIFO et fausseraient le résultat.
//!
//! Le cash (EUR, USDC, USD...) est EXCLU de l'historique : ses dépôts
//! créent des lots à coût de revient plein alors que les achats ne le
//! consomment pas toujours (jambes cash non tracées), ce qui gonfle le
//! cost basis et fabrique un P&L négatif fictif. Seuls les actifs
//! Crypto/Stock sont valorisés.
//!
//! Chaque entrée contient le snapshot de chaque actif (quantité, prix EUR,
//! valeur EUR, cost basis, fiabilité du prix) et pas seulement les totaux :
//! les totaux sont toujours la somme des actifs, et un prix suspect est
//! visible au lieu d'être noyé dans un agrégat.
//!
//! `cargo run -- --rebuild-history` recalcule tout l'historique.
//!
//! Les entrées dont la date n'est pas sur la grille de la fréquence choisie
//! (ex. un lundi quand on passe en hebdo, un dimanche en mensuel) sont
//! purgées au chargement : changer de fréquence recalcule donc l'historique
//! (les prix déjà en cache rendent ça rapide).
//!
//! Gestion jours de fermeture / crypto vs stock : entièrement déléguée à
//! `historical_price_eur` (via `portfolio_snapshot_with`).

use std::collections::HashMap;
use std::path::Path;

use anyhow::{Context, Result};
use chrono::{DateTime, Datelike, NaiveDate, TimeZone, Utc, Weekday};
use serde::{Deserialize, Serialize};

use crate::ledger::cost_basis::compute_fifo;
use crate::ledger::periods::{first_monday, is_first_monday, months_between, sundays_from};
use crate::ledger::portfolio::{portfolio_snapshot_with, AssetSnapshot, PriceFn};
use crate::market::MarketDataClient;
use crate::schema::AssetKind;
use crate::store::serialize::TxStore;

/// Fréquence des snapshots.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Frequency {
    Weekly,
    Monthly,
}

impl Frequency {
    fn label(self) -> &'static str {
        match self {
            Frequency::Weekly => "HEBDOMADAIRE (dimanche)",
            Frequency::Monthly => "MENSUEL (1er lundi)",
        }
    }

    fn is_grid_date(self, date: NaiveDate) -> bool {
        match self {
            Frequency::Weekly => date.weekday() == Weekday::Sun,
            Frequency::Monthly => is_first_monday(date),
        }
    }
}

pub const HISTORY_ENTRY_VERSION: u32 = 3;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum PriceStatus {
    #[default]
    Ok,
    Carried,
    Missing,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct HistoryAsset {
    pub symbol: String,
    pub kind: AssetKind,
    pub quantity: f64,
    pub price_eur: f64,
    pub value_eur: f64,
    pub cost_basis_eur: f64,
    #[serde(default)]
    pub price_status: PriceStatus,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct HistoryEntry {
    #[serde(alias = "week_end")]
    pub date: String,
    pub total_value_eur: f64,
    pub total_cost_basis_eur: f64,
    pub total_pnl_eur: f64,
    #[serde(default)]
    pub unrealized_pnl_eur: f64,
    #[serde(default)]
    pub realized_pnl_eur: f64,
    #[serde(default)]
    pub version: u32,
    #[serde(default)]
    pub complete: bool,
    #[serde(default)]
    pub assets: Vec<HistoryAsset>,
}

impl HistoryEntry {
    fn is_final(&self) -> bool {
        self.version >= HISTORY_ENTRY_VERSION && self.complete
    }
}

pub fn target_dates(start: NaiveDate, today: NaiveDate, freq: Frequency) -> Vec<NaiveDate> {
    match freq {
        Frequency::Weekly => weekly_dates(start, today),
        Frequency::Monthly => monthly_dates(start, today),
    }
}

fn weekly_dates(start: NaiveDate, today: NaiveDate) -> Vec<NaiveDate> {
    sundays_from(start).take_while(|d| *d < today).collect()
}

fn monthly_dates(start: NaiveDate, today: NaiveDate) -> Vec<NaiveDate> {
    months_between(start, today)
        .map(|(y, m)| first_monday(y, m))
        .filter(|d| *d >= start && *d < today)
        .collect()
}

fn retain_on_grid(history: &mut Vec<HistoryEntry>, freq: Frequency) -> usize {
    let before = history.len();
    history.retain(|e| NaiveDate::parse_from_str(&e.date, "%Y-%m-%d").map_or(false, |d| freq.is_grid_date(d)));
    before - history.len()
}

fn classify_prices(assets: &[AssetSnapshot], last_known: &HashMap<String, f64>) -> Vec<PriceStatus> {
    assets
        .iter()
        .map(|a| {
            if a.price_eur > 0.0 && a.price_eur.is_finite() {
                PriceStatus::Ok
            } else if last_known.contains_key(&a.symbol) {
                PriceStatus::Carried
            } else {
                PriceStatus::Missing
            }
        })
        .collect()
}

fn patch_prices(assets: &mut [AssetSnapshot], last_known: &mut HashMap<String, f64>) -> f64 {
    let mut total = 0.0;
    for asset in assets.iter_mut() {
        if asset.price_eur <= 0.0 {
            if let Some(&last_price) = last_known.get(&asset.symbol) {
                asset.price_eur = last_price;
                asset.value_eur = asset.quantity * last_price;
            } else {
                eprintln!("  [WARN history] prix manquant pour {}", asset.symbol);
            }
        } else {
            last_known.insert(asset.symbol.clone(), asset.price_eur);
        }
        total += asset.value_eur;
    }
    total
}

fn earliest_tx_date(tx_store: &TxStore) -> Option<NaiveDate> {
    tx_store.transactions.iter().map(|tx| tx.time.date_naive()).min()
}

fn load_history(path: &Path) -> Result<Vec<HistoryEntry>> {
    if !path.exists() {
        return Ok(Vec::new());
    }
    let raw = std::fs::read_to_string(path).with_context(|| format!("lecture de {path:?}"))?;
    match serde_json::from_str::<Vec<HistoryEntry>>(&raw) {
        Ok(history) => Ok(history),
        Err(e) => {
            let backup = path.with_extension("json.bak");
            eprintln!("[WARN history] {path:?} illisible ({e}) : copie dans {backup:?}, reconstruction complète");
            std::fs::copy(path, &backup).with_context(|| format!("sauvegarde vers {backup:?}"))?;
            Ok(Vec::new())
        }
    }
}

fn write_atomic(path: &Path, content: &str) -> Result<()> {
    if let Some(parent) = path.parent() {
        if !parent.as_os_str().is_empty() {
            std::fs::create_dir_all(parent)?;
        }
    }
    let tmp = path.with_extension("json.tmp");
    std::fs::write(&tmp, content).with_context(|| format!("écriture de {tmp:?}"))?;
    std::fs::rename(&tmp, path).with_context(|| format!("remplacement de {path:?}"))?;
    Ok(())
}

fn build_entry(
    tx_store: &TxStore,
    target: NaiveDate,
    price_fn: PriceFn,
    last_known: &mut HashMap<String, f64>,
) -> Result<HistoryEntry> {
    let at = Utc.from_utc_datetime(&target.and_hms_opt(23, 59, 59).unwrap());

    let mut snapshot = portfolio_snapshot_with(tx_store, Some(at), price_fn);
    snapshot.assets.retain(|a| a.kind != AssetKind::Cash);
    let cost_basis = compute_fifo(tx_store, Some(at))?;

    let statuses = classify_prices(&snapshot.assets, last_known);
    let total_value_eur = patch_prices(&mut snapshot.assets, last_known);

    let assets: Vec<HistoryAsset> = snapshot
        .assets
        .iter()
        .zip(statuses)
        .map(|(a, status)| HistoryAsset {
            symbol: a.symbol.clone(),
            kind: a.kind,
            quantity: a.quantity,
            price_eur: a.price_eur,
            value_eur: a.value_eur,
            cost_basis_eur: cost_basis.open_cost_basis(&a.symbol),
            price_status: status,
        })
        .collect();

    let total_cost_basis_eur: f64 = assets.iter().map(|a| a.cost_basis_eur).sum();
    let complete = assets.iter().all(|a| a.price_status == PriceStatus::Ok);

    let realized_pnl_eur = cost_basis.total_realized_pnl_stocks(tx_store);
    let unrealized_pnl_eur = total_value_eur - total_cost_basis_eur;

    Ok(HistoryEntry {
        date: target.format("%Y-%m-%d").to_string(),
        total_value_eur,
        total_cost_basis_eur,
        total_pnl_eur: unrealized_pnl_eur + realized_pnl_eur,
        unrealized_pnl_eur,
        realized_pnl_eur,
        version: HISTORY_ENTRY_VERSION,
        complete,
        assets,
    })
}

pub fn record_weekly_history(tx_store: &TxStore, path: &Path, market_client: &MarketDataClient) -> Result<()> {
    record_history(tx_store, path, Frequency::Weekly, market_client)
}

pub fn record_monthly_history(tx_store: &TxStore, path: &Path, market_client: &MarketDataClient) -> Result<()> {
    record_history(tx_store, path, Frequency::Monthly, market_client)
}

pub fn record_history(tx_store: &TxStore, path: &Path, freq: Frequency, market_client: &MarketDataClient) -> Result<()> {
    let rebuild = std::env::args().any(|a| a == "--rebuild-history");
    let live_prices = |symbol: &str, kind: AssetKind, ticker: Option<&str>, at: DateTime<Utc>| {
        market_client.historical_price_eur(symbol, at, kind.to_market_kind(), ticker)
    };
    record_history_with(tx_store, path, &live_prices, Utc::now().date_naive(), freq, rebuild)?;
    Ok(())
}

pub fn record_history_with(
    tx_store: &TxStore,
    path: &Path,
    price_fn: PriceFn,
    today: NaiveDate,
    freq: Frequency,
    rebuild: bool,
) -> Result<usize> {
    let Some(start_date) = earliest_tx_date(tx_store) else {
        return Ok(0);
    };

    let mut history = if rebuild { Vec::new() } else { load_history(path)? };
    let purged = retain_on_grid(&mut history, freq);
    let mut existing: HashMap<String, HistoryEntry> =
        history.into_iter().map(|e| (e.date.clone(), e)).collect();

    let mut last_known_prices: HashMap<String, f64> = HashMap::new();
    let mut result: Vec<HistoryEntry> = Vec::new();
    let mut computed = 0;

    for target in target_dates(start_date, today, freq) {
        let key = target.format("%Y-%m-%d").to_string();

        let entry = match existing.remove(&key) {
            Some(e) if e.is_final() => e,
            _ => {
                if computed == 0 {
                    println!("=== BACKFILL HISTORIQUE {} ===", freq.label());
                }
                let e = build_entry(tx_store, target, price_fn, &mut last_known_prices)?;
                let degraded: Vec<&str> = e
                    .assets
                    .iter()
                    .filter(|a| a.price_status != PriceStatus::Ok)
                    .map(|a| a.symbol.as_str())
                    .collect();
                if degraded.is_empty() {
                    println!("  {key} : {:.2} EUR", e.total_value_eur);
                } else {
                    println!("  {key} : {:.2} EUR  ⚠ prix manquants/reportés : {}", e.total_value_eur, degraded.join(", "));
                }
                computed += 1;
                e
            }
        };

        for a in &entry.assets {
            if a.price_eur > 0.0 {
                last_known_prices.insert(a.symbol.clone(), a.price_eur);
            }
        }
        result.push(entry);
    }

    result.extend(existing.into_values());

    if computed > 0 || purged > 0 {
        result.sort_by(|a, b| a.date.cmp(&b.date));
        write_atomic(path, &serde_json::to_string_pretty(&result)?)?;
        println!("({computed} snapshot(s) (re)calculé(s), {purged} entrée(s) hors grille purgée(s))");
    }

    let incomplete = result.iter().filter(|e| !e.complete).count();
    if incomplete > 0 {
        println!("⚠ {incomplete} snapshot(s) avec prix manquants/reportés (voir price_status) : réessayés au prochain run");
    }

    Ok(computed)
}

//! Durée de détention par actif, dérivée des lots FIFO déjà calculés
//! (`CostBasisResult`) : aucune donnée en plus, aucun appel réseau.
//!
//! - positions OUVERTES : âge des lots encore détenus à `now` ;
//! - positions FERMÉES (titres uniquement) : durée entre l'achat de chaque lot
//!   consommé et la vente. Pour la crypto, un WITHDRAW est un transfert entre
//!   wallets (vu comme une vente FIFO, voir cost_basis.rs) : le « fermé » serait
//!   trompeur, on ne le calcule pas.
//!
//! Les moyennes sont pondérées par le capital (quantité x coût unitaire) : un lot
//! de 3 000 € pèse trois fois plus qu'un lot de 1 000 €. Si le capital est nul
//! (lots à coût 0), on pondère par la quantité.

use std::collections::BTreeMap;

use chrono::{DateTime, Utc};

use crate::ledger::cost_basis::{CostBasisResult, Lot, RealizedGain};
use crate::schema::{AssetKind, ClosedSale, HoldingStats, OpenLotStat};

const SECONDS_PER_DAY: f64 = 86_400.0;
const EPSILON: f64 = 1e-12;

/// Un lot observé : sa durée de détention (jours), son capital et sa quantité.
#[derive(Debug, Clone, Copy)]
struct Sample {
    days: f64,
    cost_eur: f64,
    quantity: f64,
}

impl Sample {
    fn from_lot(lot: &Lot, until: DateTime<Utc>) -> Self {
        Self {
            days: days_between(lot.acquired_at, until),
            cost_eur: lot.quantity * lot.unit_cost_eur,
            quantity: lot.quantity,
        }
    }
}

fn days_between(from: DateTime<Utc>, to: DateTime<Utc>) -> f64 {
    ((to - from).num_seconds() as f64 / SECONDS_PER_DAY).max(0.0)
}

/// Moyenne des durées pondérée par le capital (à défaut, par la quantité).
fn weighted_avg_days(samples: &[Sample]) -> Option<f64> {
    let total_cost: f64 = samples.iter().map(|s| s.cost_eur).sum();
    let by_cost = total_cost > EPSILON;
    let weight = |s: &Sample| if by_cost { s.cost_eur } else { s.quantity };

    let total: f64 = samples.iter().map(weight).sum();
    if total <= EPSILON {
        return None;
    }
    Some(samples.iter().map(|s| s.days * weight(s)).sum::<f64>() / total)
}

fn min_days(samples: &[Sample]) -> Option<f64> {
    samples.iter().map(|s| s.days).reduce(f64::min)
}

fn max_days(samples: &[Sample]) -> Option<f64> {
    samples.iter().map(|s| s.days).reduce(f64::max)
}

#[derive(Default)]
struct Acc {
    kind: Option<AssetKind>,
    first_acquired: Option<DateTime<Utc>>,
    open: Vec<Sample>,
    open_lots: Vec<OpenLotStat>,
    open_quantity: f64,
    open_cost_eur: f64,
    closed: Vec<Sample>,
    closed_incomplete: bool,
}

impl Acc {
    fn note_acquired(&mut self, at: DateTime<Utc>) {
        self.first_acquired = Some(self.first_acquired.map_or(at, |first| first.min(at)));
    }
}

/// Les lots réellement consommés par une vente : si elle est incomplète, le dernier
/// « lot » est un reliquat fictif (coût 0, daté de la vente) qu'on écarte.
fn real_lots(gain: &RealizedGain) -> &[Lot] {
    if gain.incomplete {
        &gain.lots_consumed[..gain.lots_consumed.len().saturating_sub(1)]
    } else {
        &gain.lots_consumed[..]
    }
}

fn format_date(t: DateTime<Utc>) -> String {
    t.format("%Y-%m-%d").to_string()
}

/// Durées de détention de tous les actifs non-cash, triées par symbole.
///
/// - `kind_of` donne le type d'un symbole (ex. `|s| tx_store.assets.get(s).map(|a| a.kind)`) ;
/// - `price_of` son prix EUR actuel (ex. depuis `PortfolioSnapshot`), pour valoriser les lots ouverts.
pub fn holding_stats(
    cost_basis: &CostBasisResult,
    now: DateTime<Utc>,
    kind_of: &dyn Fn(&str) -> Option<AssetKind>,
    price_of: &dyn Fn(&str) -> Option<f64>,
) -> Vec<HoldingStats> {
    let mut by_symbol: BTreeMap<&str, Acc> = BTreeMap::new();

    for (symbol, lots) in &cost_basis.open_lots {
        let Some(kind) = kind_of(symbol).filter(|k| *k != AssetKind::Cash) else { continue };
        let acc = by_symbol.entry(symbol.as_str()).or_default();
        acc.kind = Some(kind);
        let price = price_of(symbol);
        for lot in lots {
            let sample = Sample::from_lot(lot, now);
            acc.note_acquired(lot.acquired_at);
            acc.open_quantity += lot.quantity;
            acc.open_cost_eur += sample.cost_eur;
            acc.open_lots.push(OpenLotStat {
                days: sample.days,
                cost_eur: sample.cost_eur,
                value_eur: price.map(|p| p * lot.quantity),
            });
            acc.open.push(sample);
        }
    }

    for gain in &cost_basis.realized_gains {
        if kind_of(&gain.symbol) != Some(AssetKind::Stock) {
            continue;
        }
        let acc = by_symbol.entry(gain.symbol.as_str()).or_default();
        acc.kind = Some(AssetKind::Stock);
        acc.closed_incomplete |= gain.incomplete;

        for lot in real_lots(gain) {
            acc.note_acquired(lot.acquired_at);
            acc.closed.push(Sample::from_lot(lot, gain.sell_time));
        }
    }

    by_symbol
        .into_iter()
        .map(|(symbol, acc)| HoldingStats {
            symbol: symbol.to_string(),
            kind: acc.kind.map_or("", |k| k.as_str()).to_string(),
            first_acquired: acc.first_acquired.map(format_date),
            open_quantity: acc.open_quantity,
            open_cost_eur: acc.open_cost_eur,
            open_avg_days: weighted_avg_days(&acc.open),
            open_oldest_days: max_days(&acc.open),
            open_lots: acc.open_lots,
            closed_lots: acc.closed.len() as u32,
            closed_avg_days: weighted_avg_days(&acc.closed),
            closed_min_days: min_days(&acc.closed),
            closed_max_days: max_days(&acc.closed),
            closed_incomplete: acc.closed_incomplete,
        })
        .collect()
}

/// Une ligne par vente de titres : durée de détention et résultat, pour comparer
/// la durée des gagnants à celle des perdants. Triée par date de vente.
pub fn closed_sales(
    cost_basis: &CostBasisResult,
    kind_of: &dyn Fn(&str) -> Option<AssetKind>,
) -> Vec<ClosedSale> {
    let mut sales: Vec<ClosedSale> = cost_basis
        .realized_gains
        .iter()
        .filter(|g| kind_of(&g.symbol) == Some(AssetKind::Stock))
        .map(|g| {
            let samples: Vec<Sample> =
                real_lots(g).iter().map(|lot| Sample::from_lot(lot, g.sell_time)).collect();
            ClosedSale {
                symbol: g.symbol.clone(),
                sell_date: format_date(g.sell_time),
                days: weighted_avg_days(&samples),
                cost_eur: g.cost_eur,
                proceeds_eur: g.proceeds_eur,
                pnl_eur: g.pnl_eur,
                pnl_pct: (g.cost_eur > EPSILON).then(|| g.pnl_eur / g.cost_eur * 100.0),
                incomplete: g.incomplete,
            }
        })
        .collect();
    sales.sort_by(|a, b| a.sell_date.cmp(&b.sell_date).then_with(|| a.symbol.cmp(&b.symbol)));
    sales
}

//! Arithmétique de calendrier partagée (historique, fréquence des trades).
//!
//! Évite que `trades.rs` dépende de `history.rs` juste pour `week_end`, et
//! remplace les boucles « mois par mois » / « semaine par semaine » qui
//! étaient dupliquées dans `history.rs` et `trades.rs`.

use chrono::{Datelike, Duration, NaiveDate, Weekday};

/// Dimanche de la semaine ISO contenant `date` (lundi = début de semaine).
pub fn week_end(date: NaiveDate) -> NaiveDate {
    let offset = (Weekday::Sun.num_days_from_monday() as i64
        - date.weekday().num_days_from_monday() as i64
        + 7)
        % 7;
    date + Duration::days(offset)
}

/// Premier lundi du mois `year`-`month`.
pub fn first_monday(year: i32, month: u32) -> NaiveDate {
    let first = NaiveDate::from_ymd_opt(year, month, 1).expect("mois valide");
    let offset = (7 - first.weekday().num_days_from_monday() as i64) % 7;
    first + Duration::days(offset)
}

pub fn is_first_monday(date: NaiveDate) -> bool {
    date.weekday() == Weekday::Mon && date.day() <= 7
}

/// Couples (année, mois) du mois de `start` à celui de `end`, bornes incluses.
/// Vide si `start` est dans un mois postérieur à `end`.
pub fn months_between(start: NaiveDate, end: NaiveDate) -> impl Iterator<Item = (i32, u32)> {
    let last = (end.year(), end.month());
    let first = (start.year(), start.month());
    std::iter::successors((first <= last).then_some(first), move |&(y, m)| {
        let next = if m == 12 { (y + 1, 1) } else { (y, m + 1) };
        (next <= last).then_some(next)
    })
}

/// Dimanches de fin de semaine, de la semaine de `start` à l'infini
/// (à borner avec `take_while`).
pub fn sundays_from(start: NaiveDate) -> impl Iterator<Item = NaiveDate> {
    std::iter::successors(Some(week_end(start)), |d| Some(*d + Duration::weeks(1)))
}

//! Portage de src/ledger/portfolio.py -- valorisation du portefeuille à
//! une date donnée. Combine les positions (quantités) avec les prix de
//! marché historiques.

use chrono::{DateTime, Utc};
use serde::Serialize;

use crate::ledger::positions::holdings_at;
use crate::market::MarketDataClient;
use crate::schema::AssetKind;
use crate::store::serialize::TxStore;

#[derive(Debug, Clone, Serialize)]
pub struct AssetSnapshot {
    pub symbol: String,
    pub quantity: f64,
    pub price_eur: f64,
    pub value_eur: f64,
    pub kind: AssetKind,
    pub ticker: Option<String>,
}

#[derive(Debug, Clone, Serialize)]
pub struct PortfolioSnapshot {
    pub date: String,
    pub total_value_eur: f64,
    pub assets: Vec<AssetSnapshot>,
}

/// Fonction de prix injectable : (symbole, nature, ticker, date) -> prix EUR
/// (0.0 si introuvable). Permet de tester la valorisation sans réseau.
pub type PriceFn<'a> = &'a dyn Fn(&str, AssetKind, Option<&str>, DateTime<Utc>) -> f64;

pub fn portfolio_snapshot_at(tx_store: &TxStore, at: Option<DateTime<Utc>>, market_client: &MarketDataClient) -> PortfolioSnapshot {
    let live_prices = |symbol: &str, kind: AssetKind, ticker: Option<&str>, at: DateTime<Utc>| {
        market_client.historical_price_eur(symbol, at, kind.to_market_kind(), ticker)
    };
    portfolio_snapshot_with(tx_store, at, &live_prices)
}

pub fn portfolio_snapshot_with(
    tx_store: &TxStore,
    at: Option<DateTime<Utc>>,
    price_fn: PriceFn,
) -> PortfolioSnapshot {
    let at = at.unwrap_or_else(Utc::now);

    let holdings = holdings_at(tx_store, Some(at));
    let mut total_value_eur = 0.0;
    let mut details: Vec<AssetSnapshot> = Vec::new();

    for (symbol, quantity) in holdings {
        if quantity.abs() < 1e-12 {
            continue;
        }

        let asset = tx_store.assets.get(&symbol);
        let ticker = asset.and_then(|a| a.identifiers.ticker.clone());
        let kind = asset.map(|a| a.kind).unwrap_or(AssetKind::Crypto);

        let price_eur = price_fn(&symbol, kind, ticker.as_deref(), at);
        let value_eur = quantity * price_eur;
        total_value_eur += value_eur;

        details.push(AssetSnapshot { symbol, quantity, price_eur, value_eur, kind, ticker });
    }

    details.sort_by(|a, b| b.value_eur.partial_cmp(&a.value_eur).unwrap_or(std::cmp::Ordering::Equal));

    PortfolioSnapshot { date: at.format("%Y-%m-%d").to_string(), total_value_eur, assets: details }
}

//! Portage de src/ledger/positions.py -- reconstruction des positions
//! (quantités détenues) à partir du log de transactions. Pattern
//! event-sourcing : serialized_tx.json ne stocke jamais de solde calculé,
//! seulement les transactions brutes. `holdings_at` est la seule source
//! de vérité pour "combien j'ai de X à telle date".

use std::collections::HashMap;

use chrono::{DateTime, Utc};

use crate::schema::TransactionKind;
use crate::store::serialize::TxStore;

fn sign(kind: TransactionKind) -> i8 {
    match kind {
        TransactionKind::Buy => 1,
        TransactionKind::Deposit => 1,
        TransactionKind::Sell => -1,
        TransactionKind::Withdraw => -1,
        TransactionKind::Fee => -1,
    }
}

/// Quantité détenue par symbole à la date `at` (incluse). `at = None`
/// signifie "maintenant" au sens de "toutes les transactions connues".
///
/// Les quantités quasi nulles issues d'arrondis flottants (< 1e-12) sont
/// ramenées à zéro pour éviter les faux positifs de type "je détiens
/// encore 0.0000000000003 BTC".
pub fn holdings_at(tx_store: &TxStore, at: Option<DateTime<Utc>>) -> HashMap<String, f64> {
    let mut holdings: HashMap<String, f64> = HashMap::new();

    let mut sorted: Vec<&crate::schema::Transaction> = tx_store.transactions.iter().collect();
    sorted.sort_by_key(|tx| tx.time);

    for tx in sorted {
        if let Some(at) = at {
            if tx.time > at {
                break;
            }
        }

        let symbol = tx.asset.symbol.clone();
        let s = sign(tx.kind) as f64;
        *holdings.entry(symbol).or_insert(0.0) += s * tx.quantity;
    }

    for qty in holdings.values_mut() {
        if qty.abs() < 1e-12 {
            *qty = 0.0;
        }
    }

    holdings
}

/// Comme `holdings_at`, mais filtre les actifs à quantité nulle -- pratique
/// pour n'afficher que ce qui est réellement détenu.
pub fn non_zero_holdings_at(tx_store: &TxStore, at: Option<DateTime<Utc>>) -> HashMap<String, f64> {
    holdings_at(tx_store, at)
        .into_iter()
        .filter(|(_, qty)| *qty != 0.0)
        .collect()
}

//! Plus-values réalisées par compte fiscal (CTO, PEA, Binance).
//!
//! `compute_fifo` rejoue TOUTES les transactions et indexe les lots par
//! symbole : un titre détenu à la fois sur le CTO et sur le PEA partagerait
//! ses lots entre les deux comptes, et le P&L réalisé global mélange PEA,
//! CTO et crypto. Ici on isole d'abord les transactions d'un seul compte,
//! puis on refait le FIFO sur ce périmètre.
//!
//! Périmètre actuel : les TITRES (`AssetKind::Stock`). Les cessions crypto
//! (Binance) sont calculées à part : leurs WITHDRAW (transferts) sont vus
//! comme des ventes FIFO et fausseraient le résultat.

use std::collections::BTreeMap;
use std::path::Path;

use anyhow::Result;
use chrono::Datelike;

use crate::ledger::cost_basis::compute_fifo;
use crate::schema::{AssetKind, Platform, Transaction};
use crate::store::serialize::TxStore;

/// Compte fiscal d'une transaction.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum TaxAccount {
    /// Compte-titres ordinaire (XTB, export principal `account.xlsx`).
    Cto,
    /// PEA (XTB, export `account_pea.xlsx`) : le fichier porte « pea » dans son nom.
    Pea,
    Binance,
    /// Saisies manuelles (crypto) : rattachées à aucun compte-titres.
    Other,
}

/// Compte d'une transaction : plateforme, puis nom du fichier source pour XTB
/// (les deux exports ont le même `account_label`, seul le fichier les distingue).
pub fn tax_account(tx: &Transaction) -> TaxAccount {
    match tx.platform {
        Platform::Binance => TaxAccount::Binance,
        Platform::Manual => TaxAccount::Other,
        Platform::Xtb => {
            let file = Path::new(&tx.source_file)
                .file_name()
                .map(|f| f.to_string_lossy().to_lowercase())
                .unwrap_or_default();
            // « pea » comme mot entier : account_pea.xlsx, pea.xlsx... mais pas « speaker.xlsx ».
            if file.split(|c: char| !c.is_alphanumeric()).any(|word| word == "pea") {
                TaxAccount::Pea
            } else {
                TaxAccount::Cto
            }
        }
    }
}

/// Plus-values réalisées d'un compte (en EUR).
#[derive(Debug, Default, Clone, PartialEq)]
pub struct RealizedSummary {
    /// Somme des cessions en plus-value (≥ 0).
    pub gains_eur: f64,
    /// Somme des cessions en moins-value (≤ 0).
    pub losses_eur: f64,
    /// Plus-value nette = gains + moins-values.
    pub net_eur: f64,
    /// Nombre de cessions (ventes) prises en compte.
    pub disposals: usize,
    /// Cessions dépassant les lots connus : coût inconnu traité à 0, donc plus-value surestimée.
    pub incomplete: usize,
    /// Plus-value nette par année de cession.
    pub net_by_year: BTreeMap<i32, f64>,
}

impl RealizedSummary {
    /// Texte à afficher dans le terminal.
    pub fn report(&self, title: &str) -> String {
        let mut out = format!("=== {title} ===\n");
        out.push_str(&format!(
            "Plus-value nette réalisée : {:>+.2} EUR  ({} cession{})\n",
            self.net_eur,
            self.disposals,
            if self.disposals > 1 { "s" } else { "" }
        ));
        out.push_str(&format!("  Plus-values  : {:>+12.2} EUR\n", self.gains_eur));
        out.push_str(&format!("  Moins-values : {:>+12.2} EUR\n", self.losses_eur));
        if !self.net_by_year.is_empty() {
            out.push_str("  Par année de cession :\n");
            for (year, net) in &self.net_by_year {
                out.push_str(&format!("    {year} : {net:>+12.2} EUR\n"));
            }
        }
        if self.incomplete > 0 {
            out.push_str(&format!(
                "  ⚠ {} cession(s) dépassent les lots connus (coût traité à 0) : plus-value surestimée\n",
                self.incomplete
            ));
        }
        out
    }
}

/// Plus-values réalisées sur les titres d'un compte, FIFO appliqué aux seules
/// transactions de ce compte.
pub fn realized_summary(tx_store: &TxStore, account: TaxAccount) -> Result<RealizedSummary> {
    let mut scoped = TxStore::new();
    scoped.add_transactions(
        tx_store.transactions.iter().filter(|tx| tax_account(tx) == account).cloned().collect(),
    );
    let fifo = compute_fifo(&scoped, None)?;

    let mut summary = RealizedSummary::default();
    for gain in &fifo.realized_gains {
        // Pas de cash (conversions, retraits) ni de crypto : titres uniquement.
        if scoped.assets.get(&gain.symbol).map(|a| a.kind) != Some(AssetKind::Stock) {
            continue;
        }
        if gain.pnl_eur >= 0.0 {
            summary.gains_eur += gain.pnl_eur;
        } else {
            summary.losses_eur += gain.pnl_eur;
        }
        summary.disposals += 1;
        if gain.incomplete {
            summary.incomplete += 1;
        }
        *summary.net_by_year.entry(gain.sell_time.year()).or_insert(0.0) += gain.pnl_eur;
    }
    summary.net_eur = summary.gains_eur + summary.losses_eur;
    Ok(summary)
}

//! Trades passés (Buy/Sell) et leur fréquence par mois / semaine, pour
//! dashboard.json.
//!
//! Un "trade" = une transaction Buy ou Sell sur un actif non-Cash.
//! Sont donc exclus : Deposit/Withdraw (transferts, jambes cash XTB,
//! dividendes...) et Fee.
//!
//! Attention : un Convert Binance produit deux transactions (Sell + Buy)
//! et compte donc pour 2 trades -- `buys` / `sells` dans chaque période
//! permettent de le voir.
//!
//! Les semaines sont identifiées par leur dimanche de fin (même clé que
//! history.json, pratique pour joindre les deux), les mois par "YYYY-MM".
//! Les périodes sans trade sont incluses (compteurs à 0) entre le premier
//! trade et aujourd'hui, pour que les creux soient visibles dans un graphe.

use std::collections::BTreeMap;

use chrono::{Datelike, NaiveDate, SecondsFormat, Utc};

use crate::ledger::periods::{months_between, sundays_from, week_end};
use crate::schema::{
    AssetKind, DashboardTrade, PeriodCount, TradeFrequency, Transaction, TransactionKind,
};
use crate::store::serialize::TxStore;

fn trade_txs(tx_store: &TxStore) -> impl Iterator<Item = &Transaction> {
    tx_store.transactions.iter().filter(|tx| {
        matches!(tx.kind, TransactionKind::Buy | TransactionKind::Sell)
            && tx.asset.kind != AssetKind::Cash
    })
}

/// Liste chronologique (plus ancien -> plus récent) des trades.
pub fn build_trades(tx_store: &TxStore) -> Vec<DashboardTrade> {
    let mut trades: Vec<DashboardTrade> = trade_txs(tx_store)
        .map(|tx| DashboardTrade {
            time: tx.time.to_rfc3339_opts(SecondsFormat::Secs, true),
            symbol: tx.asset.symbol.clone(),
            kind: tx.kind.as_str().to_string(),
            platform: tx.platform.as_str().to_string(),
            quantity: tx.quantity,
            // Dérivé de value_eur (autoritaire) : `price` peut être en
            // devise locale (cf. cost_basis.rs).
            unit_price_eur: if tx.quantity > 0.0 { tx.value_eur / tx.quantity } else { 0.0 },
            value_eur: tx.value_eur,
        })
        .collect();

    // RFC3339 UTC ("...Z", secondes) : l'ordre lexicographique = chronologique.
    trades.sort_by(|a, b| a.time.cmp(&b.time));
    trades
}

fn add_to_period(pc: &mut PeriodCount, is_buy: bool, value_eur: f64) {
    pc.trades += 1;
    if is_buy {
        pc.buys += 1;
    } else {
        pc.sells += 1;
    }
    pc.volume_eur += value_eur;
}

/// Nombre de trades par mois et par semaine (périodes vides incluses jusqu'à aujourd'hui).
pub fn build_trade_frequency(tx_store: &TxStore) -> TradeFrequency {
    build_trade_frequency_at(tx_store, Utc::now().date_naive())
}

/// Comme `build_trade_frequency`, avec la date du jour fournie (testable).
pub fn build_trade_frequency_at(tx_store: &TxStore, today: NaiveDate) -> TradeFrequency {
    let mut by_month: BTreeMap<(i32, u32), PeriodCount> = BTreeMap::new();
    let mut by_week: BTreeMap<NaiveDate, PeriodCount> = BTreeMap::new();
    let mut total_trades = 0u32;
    let mut first: Option<NaiveDate> = None;

    for tx in trade_txs(tx_store) {
        let date = tx.time.date_naive();
        first = Some(first.map_or(date, |f| f.min(date)));
        let is_buy = tx.kind == TransactionKind::Buy;

        let month = by_month
            .entry((date.year(), date.month()))
            .or_insert_with(|| PeriodCount::new(format!("{:04}-{:02}", date.year(), date.month())));
        add_to_period(month, is_buy, tx.value_eur);

        let we = week_end(date);
        let week = by_week
            .entry(we)
            .or_insert_with(|| PeriodCount::new(we.format("%Y-%m-%d").to_string()));
        add_to_period(week, is_buy, tx.value_eur);

        total_trades += 1;
    }

    let Some(first) = first else {
        return TradeFrequency::default(); // aucun trade
    };

    // Mois vides entre le premier trade et le mois courant.
    for (y, m) in months_between(first, today) {
        by_month
            .entry((y, m))
            .or_insert_with(|| PeriodCount::new(format!("{y:04}-{m:02}")));
    }

    // Semaines vides entre la première semaine et la semaine courante.
    let last_week = week_end(today);
    for cursor in sundays_from(first).take_while(|d| *d <= last_week) {
        by_week
            .entry(cursor)
            .or_insert_with(|| PeriodCount::new(cursor.format("%Y-%m-%d").to_string()));
    }

    let by_month: Vec<PeriodCount> = by_month.into_values().collect();
    let by_week: Vec<PeriodCount> = by_week.into_values().collect();

    TradeFrequency {
        total_trades,
        avg_per_month: total_trades as f64 / by_month.len() as f64,
        avg_per_week: total_trades as f64 / by_week.len() as f64,
        by_month,
        by_week,
    }
}

        let months: Vec<&str> = f.by_month.iter().map(|p| p.period.as_str()).collect();
        assert_eq!(months, vec!["2026-01", "2026-02", "2026-03", "2026-04"]);
        assert_eq!((f.by_month[0].buys, f.by_month[0].sells, f.by_month[0].trades), (2, 0, 2));
        assert_eq!(f.by_month[1].trades, 0); // février : creux visible
        assert_eq!((f.by_month[2].buys, f.by_month[2].sells), (0, 1));
        assert_eq!(f.by_month[0].volume_eur, 21_000.0);
        assert!((f.avg_per_month - 0.75).abs() < 1e-9); // 3 trades / 4 mois
    }

    #[test]
    fn frequency_weeks_are_consecutive_sundays_up_to_the_current_week() {
        let today = d(2026, 4, 15); // mercredi -> semaine du dimanche 19/04
        let f = build_trade_frequency_at(&fixture(), today);

        assert_eq!(f.by_week.first().unwrap().period, "2026-01-11"); // semaine du 07/01
        assert_eq!(f.by_week.last().unwrap().period, "2026-04-19");
        let dates: Vec<NaiveDate> = f
            .by_week
            .iter()
            .map(|p| NaiveDate::parse_from_str(&p.period, "%Y-%m-%d").unwrap())
            .collect();
        assert!(dates.windows(2).all(|w| (w[1] - w[0]).num_days() == 7));
        let in_week: u32 = f.by_week.iter().filter(|p| p.period == "2026-01-11").map(|p| p.trades).sum();
        assert_eq!(in_week, 2); // les deux achats du 07 et du 09/01
        assert_eq!(f.by_week.iter().map(|p| p.trades).sum::<u32>(), 3);
    }

    #[test]
    fn frequency_months_and_weeks_use_the_same_trades_total() {
        let f = build_trade_frequency_at(&fixture(), d(2026, 12, 31));

        assert_eq!(f.by_month.iter().map(|p| p.trades).sum::<u32>(), f.total_trades);
        assert_eq!(f.by_week.iter().map(|p| p.trades).sum::<u32>(), f.total_trades);
        assert_eq!(f.by_month.len(), 12);
    }
}

fn data_dir() -> PathBuf {
    PathBuf::from("./data/raw")
}

fn accounts_path() -> PathBuf {
    data_dir().join("accounts")

fn main() -> Result<()> {
    let cache_dir = PathBuf::from("./data/cache");
    
    // 1. Initialisation du client de données de marché (remplace les globals)
    let mut market_client = MarketDataClient::new();
    market_client.init_caches(&cache_dir);

    // 2. Construction du wallet (le client est passé aux parseurs pour valoriser en EUR à l'import)
    let mut tx_store = build_wallet(&market_client)?;
    warn_on_invalid_quantities(&tx_store);
    resolve_missing_tickers(&mut tx_store, &market_client);

    save_wallet(&tx_store, Path::new("./data/tx_store.json"))?;
    println!("tx_store.json regénéré.");

    // 3. Historique (le client est passé pour récupérer les prix passés)
    record_weekly_history(&tx_store, Path::new("./data/history.json"), &market_client)?;

    // 4. Valorisation actuelle et FIFO
    let snapshot = portfolio_snapshot_at(&tx_store, None, &market_client);
    let cost_basis = compute_fifo(&tx_store, None)?;

    let latent_pnl = print_valuation(&snapshot, &cost_basis);
    let realized_pnl = cost_basis.total_realized_pnl_stocks(&tx_store);
    print_performance(latent_pnl, realized_pnl);
    print_realized_by_stock(&cost_basis, &tx_store);

    // Compte-titres seul : XTB hors PEA, FIFO refait sur ce compte (voir realized.rs).
    println!();
    print!(
        "{}",
        realized_summary(&tx_store, TaxAccount::Cto)?
            .report("PLUS-VALUES RÉALISÉES : COMPTE-TITRES (XTB, hors PEA et Binance)")
    );

    // Durée de détention (lots FIFO) et ventes de titres
    let (holdings, sales) = compute_holdings(&tx_store, &cost_basis, &snapshot);

    // 5. Corrélations et Dashboard
    let correlations = compute_correlations(&tx_store, &snapshot, &market_client);
    let dashboard = build_dashboard(&tx_store, &snapshot, &cost_basis, correlations, holdings, sales);
    let dashboard_path = PathBuf::from("./data/dashboard.json");
    write_json(&dashboard_path, &dashboard)?;
    println!("Dashboard écrit : {dashboard_path:?}");

    // 6. Sauvegarde des caches sur disque avant de quitter
    market_client.save_caches(&cache_dir);
    println!("Cache des prix 1h/1j sauvegardé : {cache_dir:?}");

    devtools::dump_for_llm()
}

/// Parse toutes les sources (Binance, XTB CTO + PEA, saisies manuelles).
fn build_wallet(market_client: &MarketDataClient) -> Result<TxStore> {
    println!("=== CONSTRUCTION DU WALLET ===");
    let accounts = accounts_path();
    let mut tx_store = TxStore::new();

    // On passe le client aux parseurs qui en ont besoin (pour valoriser en EUR lors de l'import)
    tx_store.add_transactions(binance::parse_binance_sources(accounts.clone(), market_client));
    for file in ["account.xlsx", "account_pea.xlsx"] {
        tx_store.add_transactions(xtb::parse_xtb_file(&accounts.join(file), market_client));
    }
    tx_store.add_transactions(manual::parse_manual(&data_dir().join("manual_tx.json"))?);

    Ok(tx_store)
}

/// Un BUY/DEPOSIT à quantité <= 0 fait échouer le FIFO : on le signale ici.
fn warn_on_invalid_quantities(tx_store: &TxStore) {
    println!("\n=== DIAGNOSTIC : BUY/DEPOSIT à quantité négative ===");
    for tx in &tx_store.transactions {
        if matches!(tx.kind, TransactionKind::Buy | TransactionKind::Deposit) && tx.quantity <= 0.0 {
            println!(
                "  platform={:?} asset={} qty={} value_eur={} time={} source={} external_id={:?} remark={:?}",
                tx.platform, tx.asset.symbol, tx.quantity, tx.value_eur, tx.time, tx.source_file, tx.external_id, tx.remark
            );
        }
    }
}

/// Renseigne `identifiers.ticker` des actifs qui n'en ont pas (hors cash).
fn resolve_missing_tickers(tx_store: &mut TxStore, market_client: &MarketDataClient) {
    println!("\n=== RÉSOLUTION DES TICKERS ===");
    let (mut resolved, mut skipped, mut failed) = (0, 0, 0);

    let candidates: Vec<(String, AssetKind, bool)> = tx_store
        .assets
        .iter()
        .map(|(symbol, asset)| (symbol.clone(), asset.kind, asset.identifiers.ticker.is_some()))
        .collect();

    for (symbol, kind, has_ticker) in candidates {
        if kind == AssetKind::Cash || has_ticker {
            skipped += 1;
            continue;
        }
        // Utilisation de la méthode du client injecté
        match market_client.resolve_ticker(&symbol, kind.to_market_kind()) {
            Some(ticker) => {
                println!("  ✓ {symbol:<12} -> {ticker}");
                if let Some(asset) = tx_store.assets.get_mut(&symbol) {
                    asset.identifiers.ticker = Some(ticker);
                }
                resolved += 1;
            }
            None => {
                println!("  ✗ {symbol:<12} : ticker introuvable");
                failed += 1;
            }
        }
    }
    println!("({resolved} résolus, {skipped} ignorés, {failed} échoués)");
}

fn is_displayed(asset: &AssetSnapshot) -> bool {
    asset.value_eur > 0.01 && asset.kind != AssetKind::Cash && !HIDDEN_SYMBOLS.contains(&asset.symbol.as_str())
}

/// Colonnes (cost basis, P&L, P&L %) d'une ligne, et le P&L à cumuler.
fn pnl_columns(asset: &AssetSnapshot, cost_basis: &CostBasisResult) -> (String, String, String, f64) {
    if cost_basis.average_cost(&asset.symbol).is_none() {
        return (format!("{:>11}", "—"), format!("{:>11}", "—"), format!("{:>8}", "—"), 0.0);
    }
    let cb_total = cost_basis.open_cost_basis(&asset.symbol);
    let pnl_eur = asset.value_eur - cb_total;
    let pnl_pct = if cb_total > 0.0 { pnl_eur / cb_total * 100.0 } else { 0.0 };
    (format!("{cb_total:>11.2}"), format!("{pnl_eur:>+11.2}"), format!("{pnl_pct:>+7.2}%"), pnl_eur)
}

/// Tableau de valorisation. Renvoie le P&L latent total affiché.
fn print_valuation(snapshot: &PortfolioSnapshot, cost_basis: &CostBasisResult) -> f64 {
    println!("\n=== VALORISATION ACTUELLE ===");
    println!("Date: {}", snapshot.date);
    println!("Valeur totale: {:.2} EUR", snapshot.total_value_eur);
    println!("Détail par actif:");
    println!(
        "  {:<10} {:>10} {:>10} {:>12} {:>12} {:>12} {:>8}",
        "Symbole", "Qty", "Prix", "Valeur", "Cost basis", "P&L", "P&L %"
    );

    let mut total_pnl = 0.0;
    for asset in snapshot.assets.iter().filter(|a| is_displayed(a)) {
        let (cb, pnl, pct, pnl_eur) = pnl_columns(asset, cost_basis);
        total_pnl += pnl_eur;
        println!(
            "  {:<10} {:>10.4} {:>10.2} {:>12.2} {} {} {}",
            asset.symbol, asset.quantity, asset.price_eur, asset.value_eur, cb, pnl, pct
        );
    }

    println!("\nP&L latent total : {total_pnl:>+.2} EUR");
    total_pnl
}

fn print_performance(latent_pnl: f64, realized_pnl: f64) {
    println!("P&L réalisé total (actions uniquement) : {realized_pnl:>+.2} EUR");
    println!("\n=== PERFORMANCE GLOBALE (latent + réalisé actions) ===");
    println!(
        "Overall P&L : {:>+.2} EUR  (latent {latent_pnl:>+.2} + réalisé {realized_pnl:>+.2})",
        latent_pnl + realized_pnl
    );
}

fn print_realized_by_stock(cost_basis: &CostBasisResult, tx_store: &TxStore) {
    println!("\nDétail P&L réalisé par action :");
    println!("  {:<10} {:>10} {:>12} {:>12} {:>12}", "Symbole", "Qty", "Produit", "Coût", "P&L");

    // BTreeMap : symboles déjà triés.
    let mut by_symbol: BTreeMap<&str, Vec<&RealizedGain>> = BTreeMap::new();
    for gain in cost_basis.stock_realized_gains(tx_store) {
        by_symbol.entry(gain.symbol.as_str()).or_default().push(gain);
    }

    for (symbol, gains) in by_symbol {
        let qty: f64 = gains.iter().map(|g| g.quantity).sum();
        let proceeds: f64 = gains.iter().map(|g| g.proceeds_eur).sum();
        let cost: f64 = gains.iter().map(|g| g.cost_eur).sum();
        let pnl: f64 = gains.iter().map(|g| g.pnl_eur).sum();
        let flag = if gains.iter().any(|g| g.incomplete) { "  ⚠ incomplet" } else { "" };

        println!("  {symbol:<10} {qty:>10.4} {proceeds:>12.2} {cost:>12.2} {pnl:>+12.2}{flag}");
    }
}

fn compute_correlations(tx_store: &TxStore, snapshot: &PortfolioSnapshot, market_client: &MarketDataClient) -> CorrelationMatrices {
    println!("\n=== CALCUL DE LA CORRÉLATION (90j / 6m / 1an, seuil {CORRELATION_MIN_VALUE_EUR}€) ===");

    compute_correlation_matrices(tx_store, &holdings, &prices_eur, CORRELATION_MIN_VALUE_EUR, market_client)

fn dashboard_asset(asset: &AssetSnapshot, cost_basis: &CostBasisResult) -> DashboardAsset {
    let cost_basis_eur = cost_basis.open_cost_basis(&asset.symbol);
    let pnl_eur = asset.value_eur - cost_basis_eur;
    let pnl_pct = if cost_basis_eur > 0.0 { pnl_eur / cost_basis_eur * 100.0 } else { 0.0 };

    DashboardAsset {
        symbol: asset.symbol.clone(),
        kind: asset.kind.as_str().to_string(),
        ticker: asset.ticker.clone(),
        quantity: asset.quantity,
        price_eur: asset.price_eur,
        value_eur: asset.value_eur,
        cost_basis_eur,
        pnl_eur,
        pnl_pct,
        realized_pnl_eur: cost_basis.total_realized_pnl(Some(&asset.symbol)),
    }
}

fn build_dashboard(
    tx_store: &TxStore,
    snapshot: &PortfolioSnapshot,
    cost_basis: &CostBasisResult,
    correlation_matrices: CorrelationMatrices,
    holdings: Vec<HoldingStats>,
    sales: Vec<ClosedSale>,
) -> DashboardData {
    let assets: Vec<DashboardAsset> = snapshot.assets.iter().map(|a| dashboard_asset(a, cost_basis)).collect();
    let total_cost_basis_eur: f64 = assets.iter().map(|a| a.cost_basis_eur).sum();

    DashboardData {
        total_value_eur: snapshot.total_value_eur,
        total_cost_basis_eur,
        total_pnl_eur: snapshot.total_value_eur - total_cost_basis_eur,
        realized_pnl_eur: cost_basis.total_realized_pnl_stocks(tx_store),
        trades: build_trades(tx_store),
        trade_frequency: build_trade_frequency(tx_store),
        assets,
        correlation_matrices,
        holdings,
        closed_sales: sales,
    }
}

fn write_json<T: Serialize>(path: &Path, value: &T) -> Result<()> {
    fs::write(path, serde_json::to_string_pretty(value)?)?;
    Ok(())
}

fn compute_holdings(
    tx_store: &TxStore,
    cost_basis: &CostBasisResult,
    snapshot: &PortfolioSnapshot,
) -> (Vec<HoldingStats>, Vec<ClosedSale>) {
    println!("\n=== DURÉE DE DÉTENTION (FIFO) ===");
    let kind_of = |symbol: &str| tx_store.assets.get(symbol).map(|a| a.kind);
    let prices: HashMap<&str, f64> =
        snapshot.assets.iter().map(|a| (a.symbol.as_str(), a.price_eur)).collect();
    // Un prix à 0 est un prix manquant : le lot n'est alors pas valorisé.
    let price_of = |symbol: &str| prices.get(symbol).copied().filter(|p| *p > 0.0);

    let holdings = holding_stats(cost_basis, Utc::now(), &kind_of, &price_of);
    print_holdings(&holdings);
    (holdings, closed_sales(cost_basis, &kind_of))
}

fn fmt_days(days: Option<f64>) -> String {
    days.map_or("—".to_string(), |d| format!("{d:.0} j"))
}

fn print_holdings(stats: &[HoldingStats]) {
    println!(
        "  {:<10} {:>12} {:>12} {:>10} {:>10} {:>8} {:>10}",
        "Symbole", "1er achat", "Capital", "Moyenne", "Plus vieux", "Vendus", "Moy. vendus"
    );
    for s in stats {
        println!(
            "  {:<10} {:>12} {:>12.2} {:>10} {:>10} {:>8} {:>10}{}",
            s.symbol,
            s.first_acquired.as_deref().unwrap_or("—"),
            s.open_cost_eur,
            fmt_days(s.open_avg_days),
            fmt_days(s.open_oldest_days),
            s.closed_lots,
            fmt_days(s.closed_avg_days),
            if s.closed_incomplete { "  ⚠ incomplet" } else { "" }
        );
    }
}

//! Matrice de corrélation des rendements journaliers : actifs détenus
//! (au-dessus d'un seuil de valeur) + benchmarks (`benchmarks.rs`), sur des
//! fenêtres glissantes (`CORRELATION_WINDOWS`).

use std::collections::{BTreeMap, HashMap};
use chrono::{Duration, NaiveDate, Utc};

use crate::market::benchmarks::{Benchmark, BENCHMARKS};
use crate::market::MarketDataClient;
use crate::schema::AssetKind;
use crate::store::serialize::TxStore;

pub const CORRELATION_WINDOWS: &[(&str, i64)] = &[("90d", 90), ("6m", 182), ("1y", 365)];
const MIN_PAIRED_RETURNS: usize = 5;

type PriceSeries = BTreeMap<NaiveDate, f64>;

pub type CorrelationMatrix = HashMap<String, HashMap<String, Option<f64>>>;
pub type CorrelationMatrices = HashMap<String, CorrelationMatrix>;

fn to_series(points: Vec<(NaiveDate, f64)>) -> Option<PriceSeries> {
    (points.len() >= 2).then(|| points.into_iter().collect())
}

fn closes_for_portfolio_asset(
    market_client: &MarketDataClient,
    symbol: &str,
    kind: AssetKind,
    ticker: Option<&str>,
    lookback_days: i64,
) -> Option<PriceSeries> {
    let points = match kind {
        AssetKind::Crypto => market_client.binance_daily_closes(symbol, lookback_days).ok()?,
        AssetKind::Stock => market_client.yahoo_daily_closes(ticker?, lookback_days).ok()?,
        AssetKind::Cash => return None,
    };
    to_series(points)
}

fn closes_for_benchmark(market_client: &MarketDataClient, bench: &Benchmark, lookback_days: i64) -> Option<PriceSeries> {
    let points = if bench.in_eur {
        market_client.yahoo_daily_closes(bench.ticker, lookback_days)
    } else {
        market_client.yahoo_daily_closes(bench.ticker, lookback_days)
    }
    .ok()?;
    to_series(points)
}

fn paired_returns(a: &PriceSeries, b: &PriceSeries, cutoff: NaiveDate) -> Vec<(f64, f64)> {
    let mut out = Vec::new();
    let mut prev: Option<(f64, f64)> = None;

    for (date, &pa) in a.range(cutoff..) {
        let Some(&pb) = b.get(date) else { continue };
        if let Some((qa, qb)) = prev {
            if qa > 0.0 && qb > 0.0 {
                out.push((pa / qa - 1.0, pb / qb - 1.0));
            }
        }
        prev = Some((pa, pb));
    }
    out
}

fn pearson(pairs: &[(f64, f64)]) -> Option<f64> {
    if pairs.len() < MIN_PAIRED_RETURNS {
        return None;
    }

    let n = pairs.len() as f64;
    let (sum_a, sum_b) = pairs.iter().fold((0.0, 0.0), |(sa, sb), &(x, y)| (sa + x, sb + y));
    let (mean_a, mean_b) = (sum_a / n, sum_b / n);

    let (mut cov, mut var_a, mut var_b) = (0.0, 0.0, 0.0);
    for &(x, y) in pairs {
        let (dx, dy) = (x - mean_a, y - mean_b);
        cov += dx * dy;
        var_a += dx * dx;
        var_b += dy * dy;
    }

    if var_a <= 0.0 || var_b <= 0.0 {
        return None;
    }
    Some(cov / (var_a.sqrt() * var_b.sqrt()))
}

fn build_matrix(series: &[(String, PriceSeries)], cutoff: NaiveDate) -> CorrelationMatrix {
    let n = series.len();
    let mut coeffs: Vec<Vec<Option<f64>>> = vec![vec![None; n]; n];

    for i in 0..n {
        coeffs[i][i] = Some(1.0);
        for j in (i + 1)..n {
            let pairs = paired_returns(&series[i].1, &series[j].1, cutoff);
            let coeff = pearson(&pairs);
            coeffs[i][j] = coeff;
            coeffs[j][i] = coeff;
        }
    }

    series
        .iter()
        .enumerate()
        .map(|(i, (label_a, _))| {
            let row: HashMap<String, Option<f64>> =
                series.iter().enumerate().map(|(j, (label_b, _))| (label_b.clone(), coeffs[i][j])).collect();
            (label_a.clone(), row)
        })
        .collect()
}

pub fn compute_correlation_matrices(
    tx_store: &TxStore,
    holdings: &HashMap<String, f64>,
    prices_eur: &HashMap<String, f64>,
    min_value_eur: f64,
    market_client: &MarketDataClient,
) -> CorrelationMatrices {
    let max_lookback = CORRELATION_WINDOWS.iter().map(|(_, d)| *d).max().unwrap_or(365);
    let mut series: Vec<(String, PriceSeries)> = Vec::new();

    for (symbol, quantity) in holdings {
        if *quantity <= 0.0 {
            continue;
        }
        let Some(asset) = tx_store.assets.get(symbol) else { continue };
        if asset.kind == AssetKind::Cash {
            continue;
        }
        let value_eur = quantity * prices_eur.get(symbol).copied().unwrap_or(0.0);
        if value_eur < min_value_eur {
            continue;
        }
        match closes_for_portfolio_asset(market_client, symbol, asset.kind, asset.identifiers.ticker.as_deref(), max_lookback) {
            Some(prices) => series.push((symbol.clone(), prices)),
            None => eprintln!("  [WARN corrélation] historique indisponible pour {symbol}, exclu de la matrice."),
        }
    }

    for bench in BENCHMARKS {
        match closes_for_benchmark(market_client, bench, max_lookback) {
            Some(prices) => series.push((bench.label.to_string(), prices)),
            None => eprintln!(
                "  [WARN corrélation] historique indisponible pour le benchmark {} ({}), exclu.",
                bench.label, bench.ticker
            ),
        }
    }

    let today = Utc::now().date_naive();
    CORRELATION_WINDOWS
        .iter()
        .map(|(label, days)| (label.to_string(), build_matrix(&series, today - Duration::days(*days))))
        .collect()
}

//! src/market/cache.rs — cache de prix persistant (bincode), par résolution
//! horaire ou journalière. Isolé de prices.rs pour séparer "stockage" et
//! "récupération / logique métier".

use std::collections::{BTreeMap, HashMap};
use std::path::Path;
use std::sync::Mutex;

use chrono::{TimeZone, Utc};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum Resolution {
    Hour,
    Day,
}

impl Resolution {
    pub(crate) fn seconds(self) -> i64 {
        match self {
            Resolution::Hour => 3_600,
            Resolution::Day => 86_400,
        }
    }
    pub(crate) fn align(self, ts: i64) -> i64 {
        ts - ts.rem_euclid(self.seconds())
    }
}

/// True si `aligned_ts` correspond au bucket courant pour `resolution` --
/// sert à distinguer une requête "prix maintenant" d'une requête sur une
/// date passée (immuable).
#[cfg_attr(not(test), allow(dead_code))]
pub(crate) fn is_live_bucket(aligned_ts: i64, resolution: Resolution) -> bool {
    aligned_ts == resolution.align(Utc::now().timestamp())
}

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
struct PriceCache {
    data: HashMap<String, BTreeMap<i64, f64>>,
}

impl PriceCache {
    fn load(path: &Path) -> Self {
        std::fs::read(path).ok().and_then(|b| bincode::deserialize(&b).ok()).unwrap_or_default()
    }

    fn save(&self, path: &Path) -> std::io::Result<()> {
        if let Some(p) = path.parent() {
            std::fs::create_dir_all(p)?;
        }
        std::fs::write(path, bincode::serialize(self).expect("bincode serialize"))
    }

    /// Point connu le plus proche AVANT ou égal à `ts`, SANS LIMITE D'ÂGE.
    ///
    /// ATTENTION : ne pas utiliser pour valoriser un actif à une date. Le
    /// cache est creux (un point par requête passée) : le point "le plus
    /// proche avant" peut dater de plusieurs semaines/mois, et il est
    /// resservi à la place d'un vrai appel API -> prix figés dans
    /// history.json et value_eur Binance faux. Utiliser `get_exact` (prix)
    /// ou `get_within` (taux de change, avec une tolérance bornée).
    #[cfg_attr(not(test), allow(dead_code))]
    fn get_closest(&self, symbol: &str, ts: i64) -> Option<f64> {
        self.data.get(&symbol.to_uppercase())?.range(..=ts).next_back().map(|(_, &v)| v)
    }

    /// Point connu dans la fenêtre `[ts - max_age_secs, ts]` : comme
    /// `get_closest` mais un point plus vieux que `max_age_secs` est ignoré.
    fn get_within(&self, symbol: &str, ts: i64, max_age_secs: i64) -> Option<f64> {
        let lower = ts.saturating_sub(max_age_secs.max(0));
        self.data.get(&symbol.to_uppercase())?.range(lower..=ts).next_back().map(|(_, &v)| v)
    }

    /// Point connu exactement au bucket `ts`. Seul moyen sûr de savoir si
    /// on a déjà un prix FRAIS pour "maintenant".
    fn get_exact(&self, symbol: &str, ts: i64) -> Option<f64> {
        self.data.get(&symbol.to_uppercase())?.get(&ts).copied()
    }

    fn latest_ts(&self, symbol: &str) -> Option<i64> {
        self.data.get(&symbol.to_uppercase())?.keys().next_back().copied()
    }

    fn earliest_ts(&self, symbol: &str) -> Option<i64> {
        self.data.get(&symbol.to_uppercase())?.keys().next().copied()
    }

    fn range_since(&self, symbol: &str, since_ts: i64) -> Vec<(i64, f64)> {
        self.data
            .get(&symbol.to_uppercase())
            .map(|m| m.range(since_ts..).map(|(&k, &v)| (k, v)).collect())
            .unwrap_or_default()
    }

    fn insert(&mut self, symbol: &str, ts: i64, price: f64) {
        self.data.entry(symbol.to_uppercase()).or_default().insert(ts, price);
    }
}

pub(crate) struct ResolutionCache {
    resolution: Resolution,
    cache: Mutex<PriceCache>,
}

impl ResolutionCache {
    pub fn empty(resolution: Resolution) -> Self {
        Self { resolution, cache: Mutex::new(PriceCache::default()) }
    }

    pub fn init(&self, path: &Path) {
        *self.cache.lock().unwrap() = PriceCache::load(path);
    }

    pub fn save(&self, path: &Path) {
        if let Err(e) = self.cache.lock().unwrap().save(path) {
            eprintln!("Erreur sauvegarde cache: {e}");
        }
    }


    #[cfg(test)]
    pub(crate) fn get_closest(&self, symbol: &str, ts: i64) -> Option<f64> {
        self.cache.lock().unwrap().get_closest(symbol, self.resolution.align(ts))
    }

    pub(crate) fn get_within(&self, symbol: &str, ts: i64, max_age_secs: i64) -> Option<f64> {
        self.cache.lock().unwrap().get_within(symbol, self.resolution.align(ts), max_age_secs)
    }

    pub(crate) fn get_exact(&self, symbol: &str, ts: i64) -> Option<f64> {
        self.cache.lock().unwrap().get_exact(symbol, self.resolution.align(ts))
    }

    pub(crate) fn latest(&self, symbol: &str) -> Option<i64> {
        self.cache.lock().unwrap().latest_ts(symbol)
    }

    pub(crate) fn earliest(&self, symbol: &str) -> Option<i64> {
        self.cache.lock().unwrap().earliest_ts(symbol)
    }

    pub(crate) fn range_days(&self, symbol: &str, days: i64) -> Vec<(chrono::NaiveDate, f64)> {
        let cutoff = self.resolution.align((Utc::now() - chrono::Duration::days(days)).timestamp());
        self.cache
            .lock()
            .unwrap()
            .range_since(symbol, cutoff)
            .into_iter()
            .map(|(ts, price)| (Utc.timestamp_opt(ts, 0).unwrap().date_naive(), price))
            .collect()
    }

    pub(crate) fn insert(&self, symbol: &str, ts: i64, price: f64) {
        self.cache.lock().unwrap().insert(symbol, self.resolution.align(ts), price);
    }

    pub(crate) fn insert_date(&self, symbol: &str, date: chrono::NaiveDate, price: f64) {
        let ts = Utc.from_utc_datetime(&date.and_hms_opt(0, 0, 0).unwrap()).timestamp();
        self.insert(symbol, ts, price);
    }
}

        let cache = ResolutionCache::empty(Resolution::Hour);
        let ts0 = 0; // bucket 0

        assert_eq!(cache.get_exact("BTC", ts0), Some(100.0));
        // Une heure plus tard -> bucket différent -> get_exact ne doit RIEN

        assert_eq!(cache.get_exact("BTC", ts0 + hours(1)), None);
    }

        // Documente le comportement VOULU de get_closest : correct pour une

        // get_exact pour ce cas).
        let cache = ResolutionCache::empty(Resolution::Hour);
        let ts0 = 0;

        cache.insert("BTC", ts0, 100.0);

        assert_eq!(cache.get_closest("BTC", ts0 + hours(5)), Some(100.0));

    }

        cache.insert("BTC", 0, 100.0);

const BINANCE_API: &str = "https://data-api.binance.vision/api/v3/klines";
const YAHOO_CHART_API: &str = "https://query1.finance.yahoo.com/v8/finance/chart";

const FX_CACHE_TOLERANCE_SECS: i64 = 3 * 3_600;

#[derive(Debug, Error)]
pub enum PriceError {
    #[error("{0}")]
    Message(String),
    #[error(transparent)]
    Http(#[from] reqwest::Error),
}

pub fn normalize_currency_for_fx(currency: &str) -> (String, f64) {
    match currency {
        "GBp" | "GBX" => ("GBP".to_string(), 0.01),
        other => (other.to_string(), 1.0),
    }
}

fn closest_at_or_before(points: &[(i64, f64)], ts: i64) -> Option<f64> {
    points.iter().filter(|(t, _)| *t <= ts).max_by_key(|(t, _)| *t).map(|(_, p)| *p)
}

struct YahooSeries {
    points: Vec<(i64, f64)>,
    currency: String,
}

/// Service central pour les prix, taux de change et résolution de tickers.
/// Remplace les variables globales (Lazy<Mutex>) des anciens modules.
pub struct MarketDataClient {
    http_client: reqwest::blocking::Client,
    cache_1h: ResolutionCache,
    cache_1d: ResolutionCache,
    ticker_cache: Mutex<HashMap<String, Option<String>>>,
    binance_assets_cache: Mutex<Option<HashSet<String>>>,
}

impl Default for MarketDataClient {
    fn default() -> Self {
        Self::new()
    }
}

impl MarketDataClient {
    pub fn new() -> Self {
        Self {
            http_client: reqwest::blocking::Client::builder()
                .user_agent("Mozilla/5.0")
                .timeout(Duration::from_secs(10))
                .build()
                .expect("client HTTP"),
            cache_1h: ResolutionCache::empty(Resolution::Hour),
            cache_1d: ResolutionCache::empty(Resolution::Day),
            ticker_cache: Mutex::new(HashMap::new()),
            binance_assets_cache: Mutex::new(None),
        }
    }

    /// Charge les caches depuis le disque.
    pub fn init_caches(&mut self, dir: &Path) {
        self.cache_1h.init(&dir.join("price_cache_1h.bin"));
        self.cache_1d.init(&dir.join("price_cache_1d.bin"));
    }

    /// Sauvegarde les caches sur le disque.
    pub fn save_caches(&self, dir: &Path) {
        self.cache_1h.save(&dir.join("price_cache_1h.bin"));
        self.cache_1d.save(&dir.join("price_cache_1d.bin"));
    }

    // ---------------------------------------------------------------------
    // PRIX HISTORIQUES & FX
    // ---------------------------------------------------------------------

    pub fn historical_price_eur(&self, symbol: &str, time: DateTime<Utc>, kind: MarketKind, ticker: Option<&str>) -> f64 {
        let symbol = symbol.to_uppercase();
        let aligned_ts = Resolution::Hour.align(time.timestamp());

        if kind == MarketKind::Cash {
            return match symbol.as_str() {
                "EUR" | "EURI" => 1.0,
                "USD" | "USDT" | "USDC" | "BUSD" => 1.0 / self.eur_rate("USD", aligned_ts),
                "GBP" => 1.0 / self.eur_rate("GBP", aligned_ts),
                _ => 0.0,
            };
        }

        if let Some(price) = self.cache_1h.get_exact(&symbol, aligned_ts) {
            return price;
        }

        let price = match kind {
            MarketKind::Crypto => self.crypto_closes_eur(&symbol, aligned_ts * 1000, "1h", "1")
                .ok()
                .and_then(|v| v.into_iter().next())
                .map(|(_, p)| p)
                .unwrap_or(0.0),
            MarketKind::Stock => {
                let Some(ticker) = ticker else { return 0.0 };
                let day_str = Utc.timestamp_opt(aligned_ts, 0).unwrap().format("%Y-%m-%d").to_string();
                match self.yahoo_historical_price(ticker, &day_str) {
                    Ok((raw, currency)) => {
                        let (fx_currency, factor) = normalize_currency_for_fx(&currency);
                        let p = raw * factor;
                        if fx_currency != "EUR" { p / self.eur_rate(&fx_currency, aligned_ts) } else { p }
                    }
                    Err(_) => 0.0,
                }
            }
            MarketKind::Cash => unreachable!(),
        };

        if price > 0.0 {
            self.cache_1h.insert(&symbol, aligned_ts, price);
        }
        price
    }

    pub fn eur_rate(&self, currency: &str, ts: i64) -> f64 {
        let (currency, _) = normalize_currency_for_fx(currency);
        if currency == "EUR" {
            return 1.0;
        }

        let aligned_ts = Resolution::Hour.align(ts);
        let live = is_live_bucket(aligned_ts, Resolution::Hour);

        if let Some(rate) = self.lookup_fx_rate(&currency, aligned_ts, live) {
            return rate;
        }

        let pair = format!("EUR{currency}=X");
        let base = Utc.timestamp_opt(aligned_ts, 0).unwrap();
        let period1 = base - chrono::Duration::days(3);
        let period2 = (base + chrono::Duration::days(7)).min(Utc::now() + chrono::Duration::hours(1));

        if let Ok(series) = self.fetch_yahoo(&pair, period1.timestamp(), period2.timestamp(), "1h") {
            for &(t, r) in &series.points {
                if r > 0.0 {
                    self.cache_1h.insert(&currency, t, r);
                }
            }
            if let Some(rate) = closest_at_or_before(&series.points, aligned_ts) {
                if rate > 0.0 {
                    self.cache_1h.insert(&currency, aligned_ts, rate);
                    return rate;
                }
            }
        }

        eprintln!("  [WARN fx] échec {currency} ts={aligned_ts}, fallback statique");
        match currency.as_str() {
            "USD" => 1.08,
            "GBP" => 0.87,
            _ => 1.0,
        }
    }

    fn lookup_fx_rate(&self, currency: &str, aligned_ts: i64, live: bool) -> Option<f64> {
        if live {
            self.cache_1h.get_exact(currency, aligned_ts)
        } else {
            self.cache_1h.get_within(currency, aligned_ts, FX_CACHE_TOLERANCE_SECS)
        }
    }

    fn fetch_yahoo(&self, ticker: &str, period1: i64, period2: i64, interval: &str) -> Result<YahooSeries, PriceError> {
        let url = format!("{YAHOO_CHART_API}/{ticker}");
        let resp = self.http_client
            .get(&url)
            .query(&[
                ("period1", period1.to_string()),
                ("period2", period2.to_string()),
                ("interval", interval.to_string()),
                ("events", "history".to_string()),
            ])
            .send()?
            .error_for_status()?;

        let data: Value = resp.json()?;
        let result = data
            .get("chart")
            .and_then(|c| c.get("result"))
            .and_then(|r| r.as_array())
            .and_then(|a| a.first())
            .ok_or_else(|| PriceError::Message(format!("Yahoo KO pour {ticker}")))?;

        let currency = result
            .get("meta")
            .and_then(|m| m.get("currency"))
            .and_then(|c| c.as_str())
            .unwrap_or("USD")
            .to_string();

        let timestamps: Vec<i64> = result
            .get("timestamp")
            .and_then(|t| t.as_array())
            .map(|a| a.iter().filter_map(|v| v.as_i64()).collect())
            .unwrap_or_default();
        
        let closes: Vec<Option<f64>> = result
            .get("indicators")
            .and_then(|i| i.get("quote"))
            .and_then(|q| q.as_array())
            .and_then(|a| a.first())
            .and_then(|q0| q0.get("close"))
            .and_then(|c| c.as_array())
            .map(|a| a.iter().map(|v| v.as_f64()).collect())
            .unwrap_or_default();

        let points = timestamps.into_iter().zip(closes).filter_map(|(ts, c)| c.map(|c| (ts, c))).collect();

        Ok(YahooSeries { points, currency })
    }

    pub fn yahoo_historical_price(&self, ticker: &str, day_str: &str) -> Result<(f64, String), PriceError> {
        let target = chrono::NaiveDate::parse_from_str(day_str, "%Y-%m-%d").map_err(|e| PriceError::Message(e.to_string()))?;
        let target_dt = Utc.from_utc_datetime(&target.and_hms_opt(0, 0, 0).unwrap());
        let series = self.fetch_yahoo(
            ticker,
            (target_dt - chrono::Duration::days(7)).timestamp(),
            (target_dt + chrono::Duration::days(1)).timestamp(),
            "1d",
        )?;

        series
            .points
            .iter()
            .filter(|(ts, _)| Utc.timestamp_opt(*ts, 0).unwrap().date_naive() <= target)
            .max_by_key(|(ts, _)| *ts)
            .map(|(_, p)| (*p, series.currency.clone()))
            .ok_or_else(|| PriceError::Message(format!("Pas de clôture pour {ticker} au {day_str}")))
    }

    fn daily_closes_cached<F>(&self, cache_key: &str, days: i64, fetch: F) -> Result<Vec<(chrono::NaiveDate, f64)>, PriceError>
    where
        F: FnOnce(&MarketDataClient, i64) -> Result<Vec<(i64, f64)>, PriceError>,
    {
        let yesterday = Utc::now().date_naive() - chrono::Duration::days(1);
        let yesterday_ts = Resolution::Day.align(Utc.from_utc_datetime(&yesterday.and_hms_opt(0, 0, 0).unwrap()).timestamp());
        let cutoff_ts = Resolution::Day.align((Utc::now() - chrono::Duration::days(days)).timestamp());

        let fresh = self.cache_1d.latest(cache_key).map_or(false, |ts| ts >= yesterday_ts);
        let deep_enough = self.cache_1d.earliest(cache_key).map_or(false, |ts| ts <= cutoff_ts);

        if !(fresh && deep_enough) {
            let start_ts = cutoff_ts - Resolution::Day.seconds() * 5;
            for (ts, close) in fetch(self, start_ts)? {
                let day = Utc.timestamp_opt(ts, 0).unwrap().date_naive();
                self.cache_1d.insert_date(cache_key, day, close);
            }
        }

        Ok(self.cache_1d.range_days(cache_key, days))
    }

    pub fn yahoo_daily_closes(&self, ticker: &str, days: i64) -> Result<Vec<(chrono::NaiveDate, f64)>, PriceError> {
        let cache_key = format!("YHO:{ticker}");
        self.daily_closes_cached(&cache_key, days, |client, start_ts| {
            let series = client.fetch_yahoo(ticker, start_ts, Utc::now().timestamp(), "1d")?;
            let (fx_currency, price_factor) = normalize_currency_for_fx(&series.currency);
            Ok(series
                .points
                .into_iter()
                .map(|(ts, mut close)| {
                    close *= price_factor;
                    if fx_currency != "EUR" {
                        close /= client.eur_rate(&fx_currency, ts);
                    }
                    (ts, close)
                })
                .collect())
        })
    }

    fn fetch_binance_close(&self, pair: &str, start_ms: i64, interval: &str, limit: &str) -> Result<Vec<(i64, f64)>, PriceError> {
        let resp = self.http_client
            .get(BINANCE_API)
            .query(&[("symbol", pair), ("interval", interval), ("startTime", &start_ms.to_string()), ("limit", limit)])
            .send()?;
        
        if !resp.status().is_success() {
            return Ok(Vec::new()); 
        }
        
        let data: Vec<Value> = resp.json()?;
        Ok(data
            .iter()
            .filter_map(|k| {
                let ts = k.get(0)?.as_i64()? / 1000;
                let close = k.get(4)?.as_str()?.parse::<f64>().ok()?;
                Some((ts, close))
            })
            .collect())
    }

    fn crypto_closes_eur(&self, symbol: &str, start_ms: i64, interval: &str, limit: &str) -> Result<Vec<(i64, f64)>, PriceError> {
        let direct = self.fetch_binance_close(&format!("{symbol}EUR"), start_ms, interval, limit)?;
        if !direct.is_empty() {
            return Ok(direct);
        }

        let usdt = self.fetch_binance_close(&format!("{symbol}USDT"), start_ms, interval, limit)?;
        if usdt.is_empty() {
            return Ok(Vec::new());
        }
        
        let eur_usdt: HashMap<i64, f64> = self.fetch_binance_close("EURUSDT", start_ms, interval, limit)?.into_iter().collect();
        Ok(usdt.into_iter().filter_map(|(ts, p)| eur_usdt.get(&ts).map(|&fx| (ts, p / fx))).collect())
    }

    pub fn binance_daily_closes(&self, symbol: &str, days: i64) -> Result<Vec<(chrono::NaiveDate, f64)>, PriceError> {
        let cache_key = format!("BIN:{symbol}");
        self.daily_closes_cached(&cache_key, days, |client, start_ts| {
            client.crypto_closes_eur(symbol, start_ts * 1000, "1d", "1000")
        })
    }

    pub fn resolve_ticker(&self, symbol: &str, kind: MarketKind) -> Option<String> {
        match kind {
            MarketKind::Crypto => self.ticker_for_crypto(symbol),
            MarketKind::Stock => self.ticker_for_stock(symbol),
            MarketKind::Cash => None,
        }
    }

    fn ticker_for_crypto(&self, symbol: &str) -> Option<String> {
        let known = self.binance_known_base_assets();
        let symbol = symbol.to_uppercase();
        if known.contains(&symbol) {
            Some(symbol)
        } else {
            None
        }
    }

    fn binance_known_base_assets(&self) -> HashSet<String> {
        let mut cache = self.binance_assets_cache.lock().unwrap();
        if let Some(set) = cache.as_ref() {
            return set.clone();
        }

        let set = (|| -> Option<HashSet<String>> {
            let resp = self.http_client.get(BINANCE_EXCHANGE_INFO).send().ok()?.error_for_status().ok()?;
            let data: Value = resp.json().ok()?;
            let symbols = data.get("symbols")?.as_array()?;
            Some(
                symbols
                    .iter()
                    .filter_map(|s| s.get("baseAsset").and_then(|b| b.as_str()).map(String::from))
                    .collect(),
            )
        })()
        .unwrap_or_default();

        *cache = Some(set.clone());
        set
    }

    fn ticker_for_stock(&self, symbol: &str) -> Option<String> {
        let symbol = symbol.to_uppercase();

        let mut ticker_cache = self.ticker_cache.lock().unwrap();
        if let Some(cached) = ticker_cache.get(&symbol) {
            return cached.clone();
        }

        if let Some((_, over)) = manual_ticker_overrides().iter().find(|(k, _)| *k == symbol) {
            let result = Some(over.to_string());
            ticker_cache.insert(symbol, result.clone());
            return result;
        }

        let mut yahoo_symbol_guess = symbol.clone();
        let mut suffix_applied = false;
        for (xtb_suf, yahoo_suf) in xtb_to_yahoo_suffix() {
            if symbol.ends_with(xtb_suf) {
                yahoo_symbol_guess = format!("{}{}", &symbol[..symbol.len() - xtb_suf.len()], yahoo_suf);
                suffix_applied = true;
                break;
            }
        }

        if suffix_applied {
            let result = Some(yahoo_symbol_guess);
            ticker_cache.insert(symbol, result.clone());
            return result;
        }

        let queries: Vec<&str> = if yahoo_symbol_guess != symbol {
            vec![yahoo_symbol_guess.as_str(), symbol.as_str()]
        } else {
            vec![yahoo_symbol_guess.as_str()]
        };

        let mut result: Option<String> = None;
        for query in queries {
            let attempt = (|| -> Option<String> {
                let resp = self.http_client
                    .get(YAHOO_SEARCH_API)
                    .query(&[("q", query), ("quotesCount", "5"), ("newsCount", "0")])
                    .send()
                    .ok()?
                    .error_for_status()
                    .ok()?;
                let data: Value = resp.json().ok()?;
                let quotes = data.get("quotes")?.as_array()?;
                best_quote(quotes, &yahoo_symbol_guess)
            })();
            if attempt.is_some() {
                result = attempt;
                break;
            }
        }

        ticker_cache.insert(symbol, result.clone());
        result
    }
}

fn best_quote(quotes: &[Value], expected_symbol: &str) -> Option<String> {
    let candidates: Vec<&Value> = quotes
        .iter()
        .filter(|q| q.get("quoteType").and_then(|t| t.as_str()).map_or(false, |t| ACCEPTED_QUOTE_TYPES.contains(&t)))
        .collect();
    if candidates.is_empty() {
        return None;
    }

    let expected_upper = expected_symbol.to_uppercase();
    if let Some(exact) = candidates.iter().find(|q| {
        q.get("symbol").and_then(|s| s.as_str()).map(|s| s.to_uppercase()) == Some(expected_upper.clone())
    }) {
        return exact.get("symbol").and_then(|s| s.as_str()).map(String::from);
    }

    let expected_root = expected_upper.split('.').next().unwrap_or("").to_string();
    let same_root: Vec<&&Value> = candidates
        .iter()
        .filter(|q| {
            q.get("symbol")
                .and_then(|s| s.as_str())
                .map(|s| s.split('.').next().unwrap_or("").to_uppercase())
                == Some(expected_root.clone())
        })
        .collect();

    if same_root.len() == 1 {
        return same_root[0].get("symbol").and_then(|s| s.as_str()).map(String::from);
    }

    None
}

fn xtb_to_yahoo_suffix() -> &'static [(&'static str, &'static str)] {
    &[
        (".FR", ".PA"),
        (".NL", ".AS"),
        (".UK", ".L"),
        (".DE", ".DE"),
        (".US", ""),
        (".PL", ".WA"),
    ]
}

fn manual_ticker_overrides() -> &'static [(&'static str, &'static str)] {
    &[
        ("MSTR.US", "MSTR"),
        ("PRIM.US", "PRIM"),
        ("XFVT.DE", "XFVT.DE"),
        ("CNYA.DE", "CNYA.L"),
        ("STM.FR", "STM"),
    ]
}

#[derive(Debug, Default)]
pub struct TxStore {
    pub assets: HashMap<String, Asset>,
    pub transactions: Vec<Transaction>,
    pub known_external_ids: HashSet<String>,
}

impl TxStore {
    pub fn new() -> Self {
        Self::default()
    }
    pub fn is_stock(&self, symbol: &str) -> bool {
    self.assets.get(symbol).map(|a| a.kind) == Some(AssetKind::Stock)
    }
    pub fn find_or_create_asset(
        &mut self,
        symbol: &str,
        name: &str,
        kind: AssetKind,
        ref_currency: &str,
        identifiers: AssetIdentifiers,
    ) -> Asset {
        if let Some(existing) = self.assets.get(symbol) {
            return existing.clone();
        }
        let asset = Asset {
            symbol: symbol.to_string(),
            name: name.to_string(),
            kind,
            ref_currency: ref_currency.to_string(),
            identifiers,
        };
        self.assets.insert(symbol.to_string(), asset.clone());
        asset
    }

    pub fn add_transactions(&mut self, new_transactions: Vec<Transaction>) -> usize {
        let mut added = 0;
        for tx in new_transactions {
            if let Some(ext_id) = &tx.external_id {
                if self.known_external_ids.contains(ext_id) {
                    continue;
                }
            }

          
            let local_asset = self.find_or_create_asset(
                &tx.asset.symbol,
                &tx.asset.name,
                tx.asset.kind,
                &tx.asset.ref_currency,
                tx.asset.identifiers.clone(),
            );

            let ext_id = tx.external_id.clone();
            
            let mut stored_tx = tx;
            stored_tx.asset = local_asset;
            self.transactions.push(stored_tx);
            
            if let Some(ext_id) = ext_id {
                self.known_external_ids.insert(ext_id);
            }
            added += 1;
        }
        added
    }
}

fn transaction_to_dto(tx: &Transaction) -> TransactionDto {
    TransactionDto {
        platform: tx.platform,
        account_label: tx.account_label.clone(),
        kind: tx.kind,
        asset: tx.asset.symbol.clone(),
        quantity: tx.quantity,
        price: tx.price,
        value_eur: tx.value_eur,
        amount: tx.amount,
        quote_currency: tx.quote_currency.clone(),
        time: tx.time.to_rfc3339_opts(chrono::SecondsFormat::Secs, true),
        external_id: tx.external_id.clone(),
        remark: tx.remark.clone(),
        source_file: tx.source_file.clone(),
    }
}

pub fn save_wallet(tx_store: &TxStore, path: &Path) -> Result<()> {
    let assets_out: HashMap<String, AssetMeta> = tx_store
        .assets
        .iter()
        .map(|(symbol, asset)| {
            (
                symbol.clone(),
                AssetMeta {
                    name: asset.name.clone(),
                    kind: asset.kind,
                    ref_currency: asset.ref_currency.clone(),
                    identifiers: asset.identifiers.clone(),
                },
            )
        })
        .collect();

    let mut transactions_out: Vec<TransactionDto> = tx_store.transactions.iter().map(transaction_to_dto).collect();
    transactions_out.sort_by(|a, b| a.time.cmp(&b.time));

    let payload = WalletPayload {
        version: TX_STORE_VERSION,
        updated_at: Utc::now().to_rfc3339_opts(chrono::SecondsFormat::Secs, true),
        assets: assets_out,
        transactions: transactions_out,
    };

    if let Some(parent) = path.parent() {
        fs::create_dir_all(parent)?;
    }
    let json = serde_json::to_string_pretty(&payload)?;
    fs::write(path, json).with_context(|| format!("écriture de {:?}", path))?;
    Ok(())
}

use std::collections::{HashMap, HashSet};

#[derive(Debug, Error)]
pub enum BinanceParseError {
    #[error("colonne manquante: {0}")]
    MissingColumn(String),
    #[error("erreur de lecture CSV: {0}")]
    Csv(String),
    #[error("format de montant invalide '{0}'")]
    InvalidAmount(String),
    #[error("format de date invalide '{0}'")]
    InvalidTime(String),
    #[error("side inconnu: {0}")]
    UnknownSide(String),
}

// On force l'utilisation de std::result::Result pour ne pas conflit avec anyhow::Result
pub fn parse_trades(
    path: &Path, 
    known_ids: &HashSet<String>,
    price_client: &MarketDataClient,
) -> std::result::Result<Vec<Transaction>, BinanceParseError> {
    let source_file = path.display().to_string();
    let mut out = Vec::new();

    let mut reader = csv::Reader::from_path(path).map_err(|e| BinanceParseError::Csv(e.to_string()))?;
    let headers = reader.headers().map_err(|e| BinanceParseError::Csv(e.to_string()))?.clone();

    for record in reader.records() {
        let record = match record {
            Ok(r) => r,
            Err(e) => {
                eprintln!("[WARN] Ligne CSV illisible dans Binance Trades: {e}");
                continue;
            }
        };

        let row: HashMap<&str, &str> = headers.iter().zip(record.iter()).collect();
        
        macro_rules! get_col {
            ($k:expr) => {
                row.get($k).copied().ok_or_else(|| BinanceParseError::MissingColumn($k.to_string()))?
            };
        }

        let trade_id = synthetic_id(
            "binance-trade",
            &[get_col!("Time"), get_col!("Pair"), get_col!("Side"), get_col!("Price"), get_col!("Executed"), get_col!("Amount")],
        );

        if known_ids.contains(&trade_id) {
            continue;
        }

        let parsed_row = (|| -> std::result::Result<(), BinanceParseError> {
            let time_str = get_col!("Time");
            let naive = NaiveDateTime::parse_from_str(time_str, "%Y-%m-%d %H:%M:%S")
                .map_err(|_| BinanceParseError::InvalidTime(time_str.to_string()))?;
            let time = Utc.from_utc_datetime(&naive);

            let (base_qty, base_symbol) = split_amount(get_col!("Executed"))?;
            let (quote_amount, quote_symbol) = split_amount(get_col!("Amount"))?;
            let (fee_amount, fee_symbol) = split_amount(get_col!("Fee"))?;

            let side = get_col!("Side").to_uppercase();
            let kind = match side.as_str() {
                "BUY" => TransactionKind::Buy,
                "SELL" => TransactionKind::Sell,
                other => return Err(BinanceParseError::UnknownSide(other.to_string())),
            };

            let base_ref_currency = normalize_currency(&base_symbol).unwrap_or("USD").to_string();
            let base_asset = Asset {
                symbol: base_symbol.clone(),
                name: base_symbol.clone(),
                kind: asset_kind_for(&base_symbol),
                ref_currency: base_ref_currency,
                identifiers: AssetIdentifiers::default(),
            };

            let eur_price = price_client.historical_price_eur(&base_symbol, time, asset_kind_for(&base_symbol).to_market_kind(), None);
            let value_eur = base_qty * eur_price;
           
            out.push(Transaction {
                platform: Platform::Binance,
                account_label: "Spot".to_string(),
                kind,
                asset: base_asset,
                quantity: base_qty,
                price: Some(eur_price),
                amount: Some(quote_amount),
                quote_currency: Some(normalize_currency(&quote_symbol).map(String::from).unwrap_or(quote_symbol.clone())),
                time,
                value_eur,
                external_id: Some(trade_id.clone()),
                remark: None,
                source_file: source_file.clone(),
                cash_flow: None,
            });

            if fee_amount > 0.0 {
                let fee_asset = Asset {
                    symbol: fee_symbol.clone(),
                    name: fee_symbol.clone(),
                    kind: asset_kind_for(&fee_symbol),
                    ref_currency: normalize_currency(&fee_symbol).unwrap_or("USD").to_string(),
                    identifiers: AssetIdentifiers::default(),
                };
                let fee_price_eur = price_client.historical_price_eur(&fee_symbol, time, asset_kind_for(&fee_symbol).to_market_kind(), None);
                out.push(Transaction {
                    platform: Platform::Binance,
                    account_label: "Spot".to_string(),
                    kind: TransactionKind::Fee,
                    asset: fee_asset,
                    quantity: fee_amount,
                    price: Some(fee_price_eur),
                    amount: None,
                    quote_currency: None,
                    time,
                    value_eur: fee_amount * fee_price_eur,
                    external_id: None,
                    remark: Some(format!("Fee on {side} {base_symbol}")),
                    source_file: source_file.clone(),
                    cash_flow: None,
                });
            }
            Ok(())
        })();

        if let Err(e) = parsed_row {
            eprintln!("[WARN] Ligne ignorée dans Binance (ID: {}) : {}", trade_id, e);
        }
    }

    Ok(out)
}

fn split_amount(raw: &str) -> std::result::Result<(f64, String), BinanceParseError> {
    let re = Regex::new(r"^([0-9.]+)([A-Za-z]+)$").unwrap();
    let caps = re.captures(raw.trim()).ok_or_else(|| BinanceParseError::InvalidAmount(raw.to_string()))?;
    let qty: f64 = caps[1].parse().map_err(|_| BinanceParseError::InvalidAmount(raw.to_string()))?;
    Ok((qty, caps[2].to_string()))
}

pub fn synthetic_id(prefix: &str, parts: &[&str]) -> String {
    let joined = parts.join("|");
    let digest = Sha256::digest(joined.as_bytes());
    format!("{prefix}-{}", hex::encode(&digest)[..16].to_string())
}

fn split_space_amount(raw: &str) -> Result<(f64, String)> {
    let mut parts = raw.trim().splitn(2, ' ');
    let qty_str = parts.next().ok_or_else(|| anyhow!("pas de symbole dans '{raw}'"))?;
    let symbol = parts.next().ok_or_else(|| anyhow!("pas de symbole dans '{raw}'"))?;
    Ok((qty_str.parse()?, symbol.to_string()))
}

fn normalize_currency(coin: &str) -> Option<&'static str> {
    match coin {
        "EUR" | "EURI" => Some("EUR"),
        "USDC" | "BUSD" | "USD" => Some("USD"),
        _ => None,
    }
}

fn asset_kind_for(symbol: &str) -> AssetKind {
    if normalize_currency(symbol).is_some() {
        AssetKind::Cash
    } else {
        AssetKind::Crypto
    }
}

fn parse_time(raw: &str) -> Result<chrono::DateTime<Utc>> {
    let naive = NaiveDateTime::parse_from_str(raw, "%Y-%m-%d %H:%M:%S")?;
    Ok(Utc.from_utc_datetime(&naive))
}

pub fn parse_converts(
    path: &Path, 
    known_ids: &HashSet<String>,
    price_client: &MarketDataClient
) -> Result<Vec<Transaction>> {
    let source_file = path.display().to_string();
    let mut out = Vec::new();

    let mut reader = csv::Reader::from_path(path).with_context(|| format!("lecture de {path:?}"))?;
    let headers = reader.headers()?.clone();

    for record in reader.records() {
       let record = record?;
        let row: HashMap<&str, &str> = headers.iter().zip(record.iter()).collect();
        let get = |k: &str| -> Result<&str> { row.get(k).copied().ok_or_else(|| anyhow!("colonne '{k}' manquante")) };

        if get("Status")? != "Successful" { continue; }

        let convert_id = synthetic_id(
            "binance-convert",
            &[get("Time")?, get("Wallet")?, get("Pair")?, get("Sell")?, get("Buy")?, get("Price")?],
        );

        if known_ids.contains(&format!("{convert_id}-sell")) && known_ids.contains(&format!("{convert_id}-buy")) {
            continue;
        }

        let time = parse_time(get("Time")?)?;
        let (sell_qty, sell_symbol) = split_space_amount(get("Sell")?)?;
        let (buy_qty, buy_symbol) = split_space_amount(get("Buy")?)?;

        let sell_asset = Asset {
            symbol: sell_symbol.clone(),
            name: sell_symbol.clone(),
            kind: asset_kind_for(&sell_symbol),
            ref_currency: normalize_currency(&sell_symbol).unwrap_or("USD").to_string(),
            identifiers: AssetIdentifiers::default(),
        };
        let buy_asset = Asset {
            symbol: buy_symbol.clone(),
            name: buy_symbol.clone(),
            kind: asset_kind_for(&buy_symbol),
            ref_currency: normalize_currency(&buy_symbol).unwrap_or("USD").to_string(),
            identifiers: AssetIdentifiers::default(),
        };

        let sell_price_eur = price_client.historical_price_eur(&sell_symbol, time, asset_kind_for(&sell_symbol).to_market_kind(), None);
        let sell_value_eur = sell_qty * sell_price_eur;

        let buy_price_eur = price_client.historical_price_eur(&buy_symbol, time, asset_kind_for(&buy_symbol).to_market_kind(), None);
        let buy_value_eur = buy_qty * buy_price_eur;

        out.push(Transaction {
            platform: Platform::Binance,
            account_label: get("Wallet")?.to_string(),
            kind: TransactionKind::Sell,
            asset: sell_asset,
            quantity: sell_qty,
            price: Some(sell_price_eur),
            amount: Some(sell_value_eur),
            value_eur: sell_value_eur,
            quote_currency: Some("EUR".to_string()),
            time,
            external_id: Some(format!("{convert_id}-sell")),
            remark: Some("Convert".to_string()),
            source_file: source_file.clone(),
            cash_flow: None,
        });
        out.push(Transaction {
            platform: Platform::Binance,
            account_label: get("Wallet")?.to_string(),
            kind: TransactionKind::Buy,
            asset: buy_asset,
            quantity: buy_qty,
            price: Some(buy_price_eur),
            amount: Some(buy_value_eur),
            value_eur: buy_value_eur,
            quote_currency: Some("EUR".to_string()),
            time,
            external_id: Some(format!("{convert_id}-buy")),
            remark: Some("Convert".to_string()),
            source_file: source_file.clone(),
            cash_flow: None,
        });
    }

    Ok(out)
}

pub fn parse_binance_sources(accounts_path: PathBuf, price_client: &MarketDataClient) -> Vec<Transaction> {
    let mut out = Vec::new();
    let trades_path = accounts_path.join("trades.csv");
    let converts_path = accounts_path.join("convert.csv");

    if trades_path.exists() {
        println!("Lecture Binance Trades : {trades_path:?}");
        let empty_ids = HashSet::new();
        match parse_trades(&trades_path, &empty_ids, price_client) {
            Ok(mut tx) => out.append(&mut tx),
            Err(e) => eprintln!("  [Erreur Binance Trades] {e}"),
        }
    } else {
        println!("[Omis] Fichier introuvable : {trades_path:?}");
    }

    if converts_path.exists() {
        println!("Lecture Binance Converts : {converts_path:?}");
        let empty_ids = HashSet::new();
        match parse_converts(&converts_path, &empty_ids, price_client) {
            Ok(mut tx) => out.append(&mut tx),
            Err(e) => eprintln!("  [Erreur Binance Converts] {e}"),
        }
    } else {
        println!("[Omis] Fichier introuvable : {converts_path:?}");
    }

    out
}

//! Portage de src/parse/manual.py -- comble les trous non couverts par les
//! exports automatisés (ex: on-chain non tracé). Remplacé en bloc à chaque
//! run -- voir TxStore::replace_platform.

use std::fs;
use std::path::Path;

use anyhow::{anyhow, Context, Result};
use chrono::{DateTime, Utc};
use serde_json::Value;

use crate::binance::synthetic_id;
use model::schema::{Asset, AssetIdentifiers, AssetKind, Platform, Transaction, TransactionKind};

fn parse_time(raw: &str) -> Result<DateTime<Utc>> {
    Ok(DateTime::parse_from_rfc3339(&raw.replace('Z', "+00:00"))?.with_timezone(&Utc))
}

fn kind_from_str(s: &str) -> Result<TransactionKind> {
    Ok(match s {
        "Buy" => TransactionKind::Buy,
        "Sell" => TransactionKind::Sell,
        "Fee" => TransactionKind::Fee,
        "Deposit" => TransactionKind::Deposit,
        "Withdraw" => TransactionKind::Withdraw,
        other => return Err(anyhow!("TransactionKind inconnu: {other}")),
    })
}

pub fn parse_manual(path: &Path) -> Result<Vec<Transaction>> {
    if !path.exists() {
        return Ok(Vec::new());
    }

    let raw = fs::read_to_string(path).with_context(|| format!("lecture de {path:?}"))?;
    let entries: Vec<Value> = serde_json::from_str(&raw)?;
    let mut out = Vec::new();

    for entry in entries {
        let symbol = entry.get("asset").and_then(|v| v.as_str()).ok_or_else(|| anyhow!("champ 'asset' manquant"))?.to_string();
        let asset = Asset {
            symbol: symbol.clone(),
            name: symbol.clone(),
            kind: AssetKind::Crypto,
            ref_currency: "EUR".to_string(),
            identifiers: AssetIdentifiers::default(),
        };

        let raw_id = entry
            .get("id")
            .and_then(|v| v.as_str())
            .map(String::from)
            .unwrap_or_else(|| synthetic_id("manual-auto", &[&serde_json::to_string(&entry).unwrap_or_default()]));

        let kind_str = entry.get("kind").and_then(|v| v.as_str()).ok_or_else(|| anyhow!("champ 'kind' manquant"))?;

        out.push(Transaction {
            platform: Platform::Manual,
            account_label: entry.get("account_label").and_then(|v| v.as_str()).unwrap_or("Manuel").to_string(),
            kind: kind_from_str(kind_str)?,
            asset,
            quantity: entry.get("quantity").and_then(|v| v.as_f64()).ok_or_else(|| anyhow!("champ 'quantity' manquant"))?,
            price: entry.get("price").and_then(|v| v.as_f64()),
            value_eur: entry.get("value_eur").and_then(|v| v.as_f64()).ok_or_else(|| anyhow!("champ 'value_eur' manquant"))?,
            amount: entry.get("amount").and_then(|v| v.as_f64()),
            quote_currency: entry.get("quote_currency").and_then(|v| v.as_str()).map(String::from),
            time: parse_time(entry.get("time").and_then(|v| v.as_str()).ok_or_else(|| anyhow!("champ 'time' manquant"))?)?,
            external_id: Some(format!("manual-{raw_id}")),
            remark: entry.get("remark").and_then(|v| v.as_str()).map(String::from),
            source_file: path.display().to_string(),
            cash_flow: None,
        });
    }

    Ok(out)
}

#[derive(Debug, Clone)]
pub struct XtbClosedPosition {
    pub position_id: String,
    pub symbol: String,
    pub volume: f64,
    pub open_time: DateTime<Utc>,
    pub open_price: f64,
    pub close_time: DateTime<Utc>,
    pub close_price: f64,
    pub purchase_value: f64,
    pub occurrence: usize,
    pub sale_value: f64,
    pub source_file: String,
}

impl XtbClosedPosition {
    pub fn to_transactions(&self) -> Vec<Transaction> {
        let currency = infer_currency(&self.symbol);
        let asset = Asset {
            symbol: self.symbol.clone(),
            name: self.symbol.clone(),
            kind: AssetKind::Stock,
            ref_currency: currency,
            identifiers: AssetIdentifiers::default(),
        };
        vec![
            Transaction {
                platform: Platform::Xtb,
                account_label: "XTB".to_string(),
                kind: TransactionKind::Buy,
                asset: asset.clone(),
                quantity: self.volume,
                price: Some(self.open_price),
                amount: Some(self.purchase_value),
                quote_currency: None,
                time: self.open_time,
                value_eur: self.purchase_value,
                external_id: Some(self.ext_id("buy")),
                remark: None,
                source_file: self.source_file.clone(),
                cash_flow: None,
            },
            Transaction {
                platform: Platform::Xtb,
                account_label: "XTB".to_string(),
                kind: TransactionKind::Sell,
                asset,
                quantity: self.volume,
                price: Some(self.close_price),
                amount: Some(self.sale_value),
                quote_currency: None,
                time: self.close_time,
                value_eur: self.sale_value,
                external_id: Some(self.ext_id("sell")),
                remark: None,
                source_file: self.source_file.clone(),
                cash_flow: None,
            },
        ]
    }
    fn ext_id(&self, leg: &str) -> String {
        if self.occurrence == 0 {
            format!("{}-{leg}", self.position_id)
        } else {
            format!("{}#{}-{leg}", self.position_id, self.occurrence)
        }
    }
}

#[derive(Debug, Clone)]

    pub open_price: f64,
    pub purchase_value: f64,
    pub comment: Option<String>,
    pub source_file: String,
}

impl XtbOpenPosition {
    pub fn to_transaction(&self) -> Transaction {
        let currency = infer_currency(&self.symbol);
        let asset = Asset {
            symbol: self.symbol.clone(),
            name: self.symbol.clone(),
            kind: AssetKind::Stock,
            ref_currency: currency,
            identifiers: AssetIdentifiers::default(),
        };
        Transaction {
            platform: Platform::Xtb,
            account_label: "XTB".to_string(),
            kind: TransactionKind::Buy,
            asset,
            quantity: self.volume,
            price: Some(self.open_price),

            remark: self.comment.clone(),

    }
}

fn infer_currency(symbol: &str) -> String {
    let suffix = symbol.rsplit('.').next().unwrap_or("");
    match suffix {
        "DE" | "NL" | "FR" | "ES" | "IT" => "EUR",
        "UK" => "GBP",
        "US" => "USD",
        _ => "EUR",
    }
    .to_string()
}

static FX_RATE_CACHE: Lazy<Mutex<HashMap<(String, String), f64>>> = Lazy::new(|| Mutex::new(HashMap::new()));
static TRADING_CURRENCY_CACHE: Lazy<Mutex<HashMap<(String, String), String>>> = Lazy::new(|| Mutex::new(HashMap::new()));

fn fx_rate_to_eur(market_client: &MarketDataClient, currency: &str, day_str: &str) -> f64 {
    if currency == "EUR" {
        return 1.0;
    }
    let key = (currency.to_string(), day_str.to_string());
    if let Some(cached) = FX_RATE_CACHE.lock().unwrap().get(&key) {
        return *cached;
    }
    let (fx_currency, price_factor) = normalize_currency_for_fx(currency);
    let rate = match market_client.yahoo_historical_price(&format!("EUR{fx_currency}=X"), day_str) {
        Ok((fx_price, _)) if fx_price != 0.0 => price_factor / fx_price,
        _ => 1.0,
    };
    FX_RATE_CACHE.lock().unwrap().insert(key, rate);
    rate
}

fn real_trading_currency(market_client: &MarketDataClient, symbol: &str, day_str: &str) -> String {
    let key = (symbol.to_string(), day_str.to_string());
    if let Some(cached) = TRADING_CURRENCY_CACHE.lock().unwrap().get(&key) {
        return cached.clone();
    }
    let currency = market_client.resolve_ticker(symbol, AssetKind::Stock.to_market_kind())
        .and_then(|ticker| market_client.yahoo_historical_price(&ticker, day_str).ok())
        .map(|(_, currency)| currency)
        .unwrap_or_else(|| infer_currency(symbol));
    TRADING_CURRENCY_CACHE.lock().unwrap().insert(key, currency.clone());
    currency
}

pub fn find_sheet_by_prefix(path: &Path, prefix: &str) -> Result<String> {
    let workbook: Xlsx<_> = open_workbook(path).with_context(|| format!("ouverture de {path:?}"))?;
    workbook
        .sheet_names()
        .iter()
        .find(|name| name.starts_with(prefix))
        .cloned()
        .ok_or_else(|| anyhow!("aucun onglet commençant par '{prefix}'"))
}

fn cell_str(value: &Data) -> String {
    match value {
        Data::Empty => String::new(),
        Data::String(s) => s.trim().to_string(),
        Data::Float(f) => f.to_string(),
        Data::Int(i) => i.to_string(),
        Data::Bool(b) => b.to_string(),
        other => format!("{other:?}").trim().to_string(),
    }
}

fn cell_f64(value: &Data) -> Option<f64> {
    match value {
        Data::Float(f) => Some(*f),
        Data::Int(i) => Some(*i as f64),
        Data::String(s) => s.trim().parse().ok(),
        _ => None,
    }
}

fn cell_datetime(value: &Data) -> Result<DateTime<Utc>> {
    match value {
        Data::DateTime(excel_dt) => {
            let naive = excel_dt.as_datetime().ok_or_else(|| anyhow!("date Excel invalide: {excel_dt:?}"))?;
            Ok(Utc.from_utc_datetime(&naive))
        }
        Data::String(s) => {
            let naive = NaiveDateTime::parse_from_str(s.trim(), "%d/%m/%Y %H:%M:%S")
                .with_context(|| format!("format de date inattendu: {s:?}"))?;
            Ok(Utc.from_utc_datetime(&naive))
        }
        other => Err(anyhow!("format de date inattendu: {other:?}")),
    }
}

fn find_header_row(rows: &[Vec<Data>], required_col: &str) -> Result<usize> {
    let required_lower = required_col.trim().to_lowercase();
    for (i, row) in rows.iter().enumerate() {
        if row.iter().any(|cell| cell_str(cell).to_lowercase() == required_lower) {
            return Ok(i);
        }
    }
    Err(anyhow!("en-tête introuvable (colonne '{required_col}' non trouvée)"))
}

fn column_map(header_row: &[Data]) -> HashMap<String, usize> {
    header_row
        .iter()
        .enumerate()
        .filter_map(|(i, cell)| {
            let s = cell_str(cell);
            if s.is_empty() {
                None
            } else {
                Some((s.to_lowercase(), i))
            }
        })
        .collect()
}

fn col<'a>(row: &'a [Data], col_map: &HashMap<String, usize>, name: &str) -> Result<&'a Data> {
    let idx = col_map
        .get(&name.trim().to_lowercase())
        .ok_or_else(|| anyhow!("colonne '{name}' introuvable. Colonnes disponibles: {:?}", {
            let mut keys: Vec<&String> = col_map.keys().collect();
            keys.sort();
            keys
        }))?;
    row.get(*idx).ok_or_else(|| anyhow!("cellule manquante pour la colonne '{name}'"))
}

fn has_col(col_map: &HashMap<String, usize>, name: &str) -> bool {
    col_map.contains_key(&name.trim().to_lowercase())
}

fn load_rows(path: &Path, sheet_name: &str) -> Result<Vec<Vec<Data>>> {
    let mut workbook: Xlsx<_> = open_workbook(path).with_context(|| format!("ouverture de {path:?}"))?;
    let range = workbook
        .worksheet_range(sheet_name)
        .with_context(|| format!("lecture de l'onglet '{sheet_name}'"))?;
    Ok(range.rows().map(|r| r.to_vec()).collect())
}

pub fn parse_closed_positions(path: &Path, sheet_name: &str, known_ids: &std::collections::HashSet<String>) -> Result<Vec<XtbClosedPosition>> {
    let source_file = path.display().to_string();
    let rows = load_rows(path, sheet_name)?;
    let header_idx = find_header_row(&rows, "Position ID")?;
    let col_map = column_map(&rows[header_idx]);
    let mut out = Vec::new();

    let mut seen: HashMap<String, usize> = HashMap::new();
    for row in &rows[header_idx + 1..] {
        let position_id = match cell_f64(col(row, &col_map, "Position ID")?) {
            Some(id) => (id as i64).to_string(),
            None => continue,
        };
        let occurrence = {
            let n = seen.entry(position_id.clone()).or_insert(0);
            let current = *n;
            *n += 1;
            current
        };
        let buy_id = format!("{}-buy", position_id);
        let sell_id = format!("{}-sell", position_id);
        if known_ids.contains(&buy_id) && known_ids.contains(&sell_id) {
            continue;
        }
       
        out.push(XtbClosedPosition {
            position_id,
            occurrence,
            symbol: cell_str(col(row, &col_map, "Ticker")?),
            volume: cell_f64(col(row, &col_map, "Volume")?).unwrap_or(0.0),
            open_time: cell_datetime(col(row, &col_map, "Open Time (UTC)")?)?,
            open_price: cell_f64(col(row, &col_map, "Open Price")?).unwrap_or(0.0),
            close_time: cell_datetime(col(row, &col_map, "Close Time (UTC)")?)?,
            close_price: cell_f64(col(row, &col_map, "Close Price")?).unwrap_or(0.0),
            purchase_value: cell_f64(col(row, &col_map, "Purchase Value")?).unwrap_or(0.0),
            sale_value: cell_f64(col(row, &col_map, "Sale Value")?).unwrap_or(0.0),
            source_file: source_file.clone(),
        });
    }
    Ok(out)
}

pub fn parse_open_positions(path: &Path, sheet_name: &str, known_ids: &HashSet<String>, market_client: &MarketDataClient) -> Result<Vec<XtbOpenPosition>> {
    let source_file = path.display().to_string();
    let rows = load_rows(path, sheet_name)?;
    let header_idx = find_header_row(&rows, "Instrument/Position")?;
    let col_map = column_map(&rows[header_idx]);
    let mut out = Vec::new();

    for row in &rows[header_idx + 1..] {
        let side = cell_str(col(row, &col_map, "Type")?);
        if side.is_empty() { continue; }
        let position_id = cell_str(col(row, &col_map, "Instrument/Position")?);
        if position_id.is_empty() { continue; }

        if known_ids.contains(&position_id) {
            continue;
        }

        let symbol = cell_str(col(row, &col_map, "Ticker")?);
        let open_time = cell_datetime(col(row, &col_map, "Open time (UTC)")?)?;
        let open_price = cell_f64(col(row, &col_map, "Open price")?).unwrap_or(0.0);
        let volume = cell_f64(col(row, &col_map, "Volume")?).unwrap_or(0.0);

        let day_str = open_time.format("%Y-%m-%d").to_string();
        let currency = real_trading_currency(market_client, &symbol, &day_str);
        let fx_rate = fx_rate_to_eur(market_client, &currency, &day_str);
        let acquisition_cost_eur = volume * open_price * fx_rate;

        out.push(XtbOpenPosition {
            position_id,
            symbol,
            volume,
            open_time,
            open_price,
            purchase_value: acquisition_cost_eur,
            comment: None, // colonne "Comment" absente du nouvel export
            source_file: source_file.clone(),
        });
    }

    Ok(out)
}

fn cash_kind_for(op_type: &str) -> Option<(TransactionKind, CashFlowKind)> {
    use CashFlowKind::*;
    use TransactionKind as K;
    match op_type {
        "Deposit" => Some((K::Deposit, ExternalDeposit)),
        "Withdrawal" => Some((K::Withdraw, ExternalWithdrawal)), // nom à confirmer sur ton export
        "Free funds interest" | "Dividend" => Some((K::Deposit, Income)),
        "Stock purchase" => Some((K::Withdraw, TradeLeg)),
        "Stock sell" | "Fractional shares" => Some((K::Deposit, TradeLeg)),
        "Withholding tax" | "Tax IFTT" => Some((K::Fee, Tax)),
        _ => None,
    }
}

const SKIP_CASH_TYPES: &[&str] = &["PEA deposit"];

pub fn parse_cash_operations(path: &Path, sheet_name: &str) -> Result<Vec<Transaction>> {
    let source_file = path.display().to_string();
    let rows = load_rows(path, sheet_name)?;
    let header_idx = find_header_row(&rows, "ID")?;
    let col_map = column_map(&rows[header_idx]);
    let mut out = Vec::new();

    let eur_asset = Asset {
        symbol: "EUR".to_string(),
        name: "EUR".to_string(),
        kind: AssetKind::Cash,
        ref_currency: "EUR".to_string(),
        identifiers: AssetIdentifiers::default(),
    };

    for row in &rows[header_idx + 1..] {
        let op_type = cell_str(col(row, &col_map, "Type")?);
        if op_type.is_empty() || op_type == "Total" {
            continue;
        }
        if SKIP_CASH_TYPES.contains(&op_type.as_str()) {
            continue;
        }

        let (kind, cash_flow) = cash_kind_for(&op_type)
    .ok_or_else(|| anyhow!("Type d'opération Cash non géré: {op_type:?}"))?;
        let amount = cell_f64(col(row, &col_map, "Amount")?).unwrap_or(0.0);
        let op_id = cell_str(col(row, &col_map, "ID")?);
        let comment = if has_col(&col_map, "Comment") { cell_str(col(row, &col_map, "Comment")?) } else { String::new() };

        out.push(Transaction {
            platform: Platform::Xtb,
            account_label: "XTB".to_string(),
            kind: kind,
            cash_flow: Some(cash_flow),
            asset: eur_asset.clone(),
            quantity: amount.abs(),
            price: Some(1.0),
            value_eur: amount.abs(),
            amount: Some(amount),
            quote_currency: Some("EUR".to_string()),
            time: cell_datetime(col(row, &col_map, "Time")?)?,
            external_id: Some(format!("xtb-cash-{op_id}")),
            remark: if comment.is_empty() { None } else { Some(comment) },
            source_file: source_file.clone(),
        });
    }

    Ok(out)
}

pub fn parse_xtb_file(path: &Path, market_client: &MarketDataClient) -> Vec<Transaction> {
    let mut out = Vec::new();
    if !path.exists() {
        println!("[Omis] Fichier introuvable : {path:?}");
        return out;
    }

    println!("Lecture XTB : {path:?}");
    let empty_ids = HashSet::new();

    match find_sheet_by_prefix(path, "Closed Position") {
        Ok(sheet) => match parse_closed_positions(path, &sheet, &empty_ids) {
            Ok(positions) => {
                for pos in positions {
                    out.extend(pos.to_transactions());
                }
            }
            Err(e) => println!("  [Erreur XTB Closed] {e}"),
        },
        Err(e) => println!("  [Erreur XTB Closed] {e}"),
    }

    match find_sheet_by_prefix(path, "Open Position") {
        Ok(sheet) => match parse_open_positions(path, &sheet, &empty_ids, market_client) {
            Ok(positions) => out.extend(positions.iter().map(|p| p.to_transaction())),
            Err(e) => println!("  [Erreur XTB Open] {e}"),
        },
        Err(e) => println!("  [Erreur XTB Open] {e}"),
    }

    match find_sheet_by_prefix(path, "Cash") {
        Ok(sheet) => match parse_cash_operations(path, &sheet) {
            Ok(mut tx) => out.append(&mut tx),
            Err(e) => println!("  [Erreur XTB Cash] {e}"),
        },
        Err(e) => println!("  [Erreur XTB Cash] {e}"),
    }

    out
}

#[cfg(test)]
mod tests {
    use super::*;
    use chrono::TimeZone;
    use rust_xlsxwriter::{ExcelDateTime, Format, Workbook};

    fn utc(y: i32, mo: u32, d: u32, h: u32, mi: u32, s: u32) -> DateTime<Utc> {
        Utc.with_ymd_and_hms(y, mo, d, h, mi, s).unwrap()
    }

    /// Chemin de fichier temporaire unique par test (pas de dépendance `tempfile`).
    fn temp_path(name: &str) -> std::path::PathBuf {
        std::env::temp_dir().join(format!("importers-xtb-{}-{name}.xlsx", std::process::id()))
    }

    enum When {
        Text(&'static str),
        Excel(u16, u8, u8, u16, u8, u8),
    }

    /// Écrit un export XTB minimal : une ligne de préambule, l'en-tête, puis les opérations.
    fn write_cash_workbook(path: &Path, sheet: &str, ops: &[(&str, &str, f64, When, &str)]) {
        let mut workbook = Workbook::new();
        let date_format = Format::new().set_num_format("dd/mm/yyyy hh:mm:ss");
        let ws = workbook.add_worksheet();
        ws.set_name(sheet).unwrap();
        ws.write_string(0, 0, "Account").unwrap();
        ws.write_string(0, 1, "123456").unwrap();
        for (c, h) in ["ID", "Type", "Time", "Amount", "Comment"].iter().enumerate() {
            ws.write_string(2, c as u16, *h).unwrap();
        }
        for (i, (id, kind, amount, when, comment)) in ops.iter().enumerate() {
            let row = 3 + i as u32;
            ws.write_string(row, 0, *id).unwrap();
            ws.write_string(row, 1, *kind).unwrap();
            match when {
                When::Text(s) => {
                    ws.write_string(row, 2, *s).unwrap();
                }
                When::Excel(y, mo, d, h, mi, s) => {
                    let dt = ExcelDateTime::from_ymd(*y, *mo, *d).unwrap().and_hms(*h, *mi, *s).unwrap();
                    ws.write_datetime_with_format(row, 2, &dt, &date_format).unwrap();
                }
            }
            ws.write_number(row, 3, *amount).unwrap();
            if !comment.is_empty() {
                ws.write_string(row, 4, *comment).unwrap();
            }
        }
        workbook.save(path).unwrap();
    }

    #[test]
    fn cash_operations_are_read_from_a_real_xlsx_file() {
        let path = temp_path("cash-ok");
        write_cash_workbook(
            &path,
            "Cash Operations 2025",
            &[
                ("1001", "Deposit", 1000.0, When::Text("01/03/2025 10:00:00"), "virement"),
                ("1002", "Dividend", 12.5, When::Excel(2025, 3, 15, 9, 30, 0), ""),
                ("1003", "Stock purchase", -500.0, When::Text("16/03/2025 11:00:00"), ""),
                ("1004", "Withholding tax", -3.2, When::Text("16/03/2025 11:05:00"), ""),
                ("1005", "PEA deposit", 200.0, When::Text("17/03/2025 08:00:00"), ""), // ignoré volontairement
                ("", "Total", 0.0, When::Text("17/03/2025 08:00:00"), ""),              // ligne de total ignorée
            ],
        );

        let sheet = find_sheet_by_prefix(&path, "Cash").unwrap();
        assert_eq!(sheet, "Cash Operations 2025");
        let tx = parse_cash_operations(&path, &sheet).unwrap();
        let _ = std::fs::remove_file(&path);

        assert_eq!(tx.len(), 4);
        let kinds: Vec<_> = tx.iter().map(|t| t.kind).collect();
        assert_eq!(kinds, [TransactionKind::Deposit, TransactionKind::Deposit, TransactionKind::Withdraw, TransactionKind::Fee]);
        let flows: Vec<_> = tx.iter().map(|t| t.cash_flow).collect();
        assert_eq!(
            flows,
            [
                Some(CashFlowKind::ExternalDeposit),
                Some(CashFlowKind::Income),
                Some(CashFlowKind::TradeLeg),
                Some(CashFlowKind::Tax)
            ]
        );

        // quantité et valeur toujours positives, montant signé conservé
        assert_eq!(tx[2].quantity, 500.0);
        assert_eq!(tx[2].value_eur, 500.0);
        assert_eq!(tx[2].amount, Some(-500.0));
        assert_eq!(tx[3].value_eur, 3.2);

        assert_eq!(tx[0].external_id.as_deref(), Some("xtb-cash-1001"));
        assert_eq!(tx[0].time, utc(2025, 3, 1, 10, 0, 0));
        assert_eq!(tx[1].time, utc(2025, 3, 15, 9, 30, 0)); // cellule date Excel (calamine `dates`)
        assert_eq!(tx[0].remark.as_deref(), Some("virement"));
        assert_eq!(tx[1].remark, None);

        assert!(tx.iter().all(|t| t.platform == Platform::Xtb && t.asset.symbol == "EUR" && t.asset.kind == AssetKind::Cash));
    }

    #[test]
    fn an_unknown_cash_operation_type_fails_the_whole_sheet_with_a_clear_message() {
        let path = temp_path("cash-unknown");
        write_cash_workbook(

            "Cash Operations",

    }

    }

    #[test]
    fn cash_operation_types_map_to_a_kind_and_a_cash_flow() {
        use CashFlowKind::*;
        use TransactionKind as K;
        assert_eq!(cash_kind_for("Deposit"), Some((K::Deposit, ExternalDeposit)));
        assert_eq!(cash_kind_for("Withdrawal"), Some((K::Withdraw, ExternalWithdrawal)));
        assert_eq!(cash_kind_for("Free funds interest"), Some((K::Deposit, Income)));
        assert_eq!(cash_kind_for("Dividend"), Some((K::Deposit, Income)));
        assert_eq!(cash_kind_for("Stock purchase"), Some((K::Withdraw, TradeLeg)));
        assert_eq!(cash_kind_for("Stock sell"), Some((K::Deposit, TradeLeg)));
        assert_eq!(cash_kind_for("Fractional shares"), Some((K::Deposit, TradeLeg)));
        assert_eq!(cash_kind_for("Withholding tax"), Some((K::Fee, Tax)));
        assert_eq!(cash_kind_for("Tax IFTT"), Some((K::Fee, Tax)));
        assert_eq!(cash_kind_for("PEA deposit"), None); // géré à part (SKIP_CASH_TYPES)
        assert!(SKIP_CASH_TYPES.contains(&"PEA deposit"));
        assert_eq!(cash_kind_for("anything else"), None);
    }

    #[test]
    fn cell_helpers_convert_the_cell_types_calamine_produces() {
        assert_eq!(cell_str(&Data::Empty), "");
        assert_eq!(cell_str(&Data::String("  Deposit ".into())), "Deposit");
        assert_eq!(cell_str(&Data::Int(42)), "42");
        assert_eq!(cell_f64(&Data::Float(1.5)), Some(1.5));
        assert_eq!(cell_f64(&Data::Int(2)), Some(2.0));
        assert_eq!(cell_f64(&Data::String(" 3.25 ".into())), Some(3.25));
        assert_eq!(cell_f64(&Data::String("abc".into())), None);
        assert_eq!(cell_f64(&Data::Empty), None);
        assert_eq!(cell_datetime(&Data::String("01/03/2025 10:00:00".into())).unwrap(), utc(2025, 3, 1, 10, 0, 0));
        assert!(cell_datetime(&Data::String("2025-03-01".into())).is_err());
        assert!(cell_datetime(&Data::Empty).is_err());
    }

    #[test]
    fn header_row_and_columns_are_found_case_insensitively() {
        let rows = vec![
            vec![Data::String("Account".into()), Data::String("123".into())],
            vec![Data::String("id".into()), Data::String("TYPE".into()), Data::Empty, Data::String("Amount".into())],
        ];
        assert_eq!(find_header_row(&rows, "ID").unwrap(), 1);
        assert!(find_header_row(&rows, "Symbol").is_err());

        let map = column_map(&rows[1]);
        assert_eq!(map.get("type"), Some(&1));
        assert!(!map.contains_key("")); // les en-têtes vides sont ignorés
        let data_row = vec![Data::String("7".into()), Data::String("Deposit".into()), Data::Empty, Data::Float(9.0)];
        assert_eq!(cell_str(col(&data_row, &map, " Type ").unwrap()), "Deposit");
        let err = col(&data_row, &map, "Nope").unwrap_err().to_string();
        assert!(err.contains("Nope") && err.contains("amount"), "message inattendu : {err}");
        assert!(has_col(&map, "AMOUNT") && !has_col(&map, "Comment"));
    }
